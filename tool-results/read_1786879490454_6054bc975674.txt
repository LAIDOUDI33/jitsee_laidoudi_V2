    78→function parseSizeToBytes(size: string): number {
    79→  const match = size.match(/([\d.]+)\s*(KB|MB|GB)/)
    80→  if (!match) return 0
    81→  const val = parseFloat(match[1])
    82→  if (match[2] === 'GB') return val * 1024 * 1024 * 1024
    83→  if (match[2] === 'MB') return val * 1024 * 1024
    84→  return val * 1024
    85→}
    86→
    87→function formatFileSize(bytes: number): string {
    88→  if (bytes === 0) return '0 B'
    89→  const units = ['B', 'KB', 'MB', 'GB']
    90→  const i = Math.floor(Math.log(bytes) / Math.log(1024))
    91→  const val = bytes / Math.pow(1024, i)
    92→  return `${val < 10 ? val.toFixed(1) : Math.round(val)} ${units[i]}`
    93→}
