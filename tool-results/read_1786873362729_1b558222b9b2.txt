   464→          <div className='flex items-start gap-3'>
   465→            {/* Unread dot + Icon container */}
   466→            <div className='relative shrink-0 mt-0.5'>
   467→              {notification.unread && (

   477→              <div className='flex items-start justify-between gap-2'>
   478→                <div className='flex items-center gap-2 min-w-0'>
   479→                  <div className={`w-6 h-6 rounded-full flex items-center justify-center text-[10px] font-bold shrink-0 ${notification.sender.color}`}>
   480→                    {notification.sender.initials}
   481→                  </div>
   482→                  <span className='text-xs text-muted-foreground truncate'>{notification.sender.name}</span>
   483→                </div>
   484→                <div className='flex items-center gap-1 shrink-0'>
   485→                  {notification.pinned && (
   486→                    <Pin className='h-3 w-3 text-amber-500' />
   487→                  )}
   488→                  <span className='text-[11px] text-muted-foreground whitespace-nowrap'>{notification.timestamp}</span>
   489→                </div>
   490→              </div>
   491→              <p
   492→                className='text-sm font-medium mt-1 leading-snug'
   493→                dangerouslySetInnerHTML={{ __html: notification.title }}
   494→              />
   495→              <p
   496→                className='text-xs text-muted-foreground mt-0.5 leading-relaxed line-clamp-2'
   497→                dangerouslySetInnerHTML={{ __html: notification.description }}
   498→              />
   499→
   500→              {/* Expanded detail */}
   501→              <AnimatePresence>
   502→                {expanded && (
   503→                  <motion.div
   504→                    initial={{ height: 0, opacity: 0 }}
   505→                    animate={{ height: 'auto', opacity: 1 }}
   506→                    exit={{ height: 0, opacity: 0 }}
   507→                    transition={{ duration: 0.25, ease: 'easeInOut' }}
   508→                    className='overflow-hidden'
   509→                  >
   510→                    <div className='mt-3 pt-3 border-t border-border/50'>
   511→                      <p className='text-xs text-muted-foreground leading-relaxed'>{notification.detail}</p>
   512→                      {notification.actions && (
   513→                        <div className='flex items-center gap-2 mt-3' onClick={e => e.stopPropagation()}>
   514→                          {notification.actions.map((action) => (
   515→                            <Button
   516→                              key={action.label}
   517→                              size='sm'
   518→                              variant={action.variant}
   519→                              className='h-7 text-xs'
   520→                              onClick={() => onAction(notification.id, action.onClickLabel)}
   521→                            >
   522→                              {action.variant === 'default' && <CheckCircle2 className='h-3 w-3 mr-1' />}
   523→                              {action.variant === 'outline' && <XCircle className='h-3 w-3 mr-1' />}
   524→                              {action.label}
   525→                            </Button>
   526→                          ))}
   527→                        </div>
   528→                      )}
   529→                    </div>
   530→                  </motion.div>
   531→                )}
   532→              </AnimatePresence>
   533→            </div>
   534→
   535→            {/* Action buttons */}
   536→            <div className='flex flex-col items-center gap-1 shrink-0 ml-1' onClick={e => e.stopPropagation()}>
   537→              <button
   538→                onClick={() => onTogglePin(notification.id)}
   539→                className='p-1 rounded-md hover:bg-muted transition-colors duration-150 text-muted-foreground hover:text-foreground'
   540→                title={notification.pinned ? 'Unpin' : 'Pin'}
   541→              >
   542→                {notification.pinned ? <PinOff className='h-3.5 w-3.5' /> : <Pin className='h-3.5 w-3.5' />}
   543→              </button>
   544→              <button
   545→                onClick={() => onDelete(notification.id)}
   546→                className='p-1 rounded-md hover:bg-red-500/10 transition-colors duration-150 text-muted-foreground hover:text-red-500'
   547→                title='Delete'
   548→              >
   549→                <Trash2 className='h-3.5 w-3.5' />
   550→              </button>
   551→            </div>
   552→          </div>
   553→        </CardContent>
   554→      </Card>
   555→    </motion.div>
   556→  )
   557→}

   588→
   589→  // Count unread
   590→  const unreadCount = useMemo(
   591→    () => notifications.filter(n => n.unread).length,
   592→    [notifications]

   594→
   595→  // Filter by category (filter pills)
   596→  const categoryFiltered = useMemo(() => {
   597→    switch (activeFilter) {
   598→      case 'unread':
   599→        return notifications.filter(n => n.unread)
   600→      case 'mentions':
   601→        return notifications.filter(n => n.type === 'mention')
   602→      case 'meetings':
   603→        return notifications.filter(n => ['meeting-invite', 'meeting-soon', 'recording-ready'].includes(n.type))
   604→      case 'system':
   605→        return notifications.filter(n => ['security-alert', 'maintenance', 'system-update'].includes(n.type))
   606→      default:
   607→        return notifications
   608→    }
   609→  }, [notifications, activeFilter])
   610→
   611→  // Filter by tab
   612→  const tabFiltered = useMemo(() => {
   613→    switch (activeTab) {
   614→      case 'mentions':
   615→        return categoryFiltered.filter(n => n.type === 'mention')
   616→      case 'meeting-alerts':
   617→        return categoryFiltered.filter(n => ['meeting-invite', 'meeting-soon', 'recording-ready', 'ai-summary'].includes(n.type))
   618→      case 'system-updates':
   619→        return categoryFiltered.filter(n => ['security-alert', 'maintenance', 'system-update', 'member-joined'].includes(n.type))
   620→      default:
   621→        return categoryFiltered
   622→    }
   623→  }, [categoryFiltered, activeTab])
   624→
   625→  // Sort: pinned first, then by time group, then by original order
   626→  const sorted = useMemo(() => {
   627→    const groupOrder: Record<TimeGroup, number> = { 'Today': 0, 'Yesterday': 1, 'Earlier': 2 }
   628→    return [...tabFiltered].sort((a, b) => {
   629→      if (a.pinned !== b.pinned) return a.pinned ? -1 : 1
   630→      if (groupOrder[a.timeGroup] !== groupOrder[b.timeGroup]) {
   631→        return groupOrder[a.timeGroup] - groupOrder[b.timeGroup]
   632→      }
   633→      return 0
   634→    })
   635→  }, [tabFiltered])
   636→
   637→  // Group by time for 'recent' tab
   638→  const grouped = useMemo(() => {
   639→    if (activeTab !== 'recent') return { 'All': sorted }
   640→    const groups: Record<TimeGroup, Notification[]> = { 'Today': [], 'Yesterday': [], 'Earlier': [] }
   641→    sorted.forEach(n => groups[n.timeGroup].push(n))
