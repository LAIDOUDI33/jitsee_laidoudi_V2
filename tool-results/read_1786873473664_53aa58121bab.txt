    84→function useCountUp(target: number, duration = 1200, delay = 0) {
    85→  const [count, setCount] = useState(0)
    86→  useEffect(() => {
    87→    const startTime = performance.now() + delay
    88→    function step(now: number) {
    89→      if (now < startTime) { requestAnimationFrame(step); return }
    90→      const elapsed = now - startTime
    91→      const progress = Math.min(elapsed / duration, 1)
    92→      const eased = 1 - Math.pow(1 - progress, 3)
    93→      setCount(Math.round(eased * target))
    94→      if (progress < 1) requestAnimationFrame(step)
    95→    }
    96→    requestAnimationFrame(step)
    97→  }, [target, duration, delay])
    98→  return count
    99→}
