   262→// ─── Network Quality Indicator ────────────────────────────────
   263→function NetworkQualityIndicator() {
   264→  const [quality, setQuality] = useState<NetworkQuality>('excellent');
   265→  const [latency, setLatency] = useState(22);
   266→
   267→  useEffect(() => {
   268→    const qualities: NetworkQuality[] = ['excellent', 'good', 'fair', 'poor'];
   269→    const changeQuality = () => {
   270→      const q = qualities[Math.floor(Math.random() * qualities.length)];
   271→      setQuality(q);
   272→      const [minLat, maxLat] = networkQualityConfig[q].latency;
   273→      setLatency(Math.floor(Math.random() * (maxLat - minLat + 1)) + minLat);
   274→    };
   275→    const interval = setInterval(changeQuality, 10000 + Math.random() * 5000);
   276→    return () => clearInterval(interval);
   277→  }, []);
   278→
   279→  const config = networkQualityConfig[quality];
   280→  const barCount = quality === 'excellent' ? 4 : quality === 'good' ? 3 : quality === 'fair' ? 2 : 1;
   281→
   282→  return (
   283→    <Tooltip>
   284→      <TooltipTrigger asChild>
   285→        <div className="absolute top-4 right-4 z-30 flex items-center gap-2 px-2.5 py-1.5 rounded-xl bg-black/40 backdrop-blur-xl border border-white/10 cursor-default">
   286→          <div className="flex items-end gap-[2px] h-3.5">
   287→            {[1, 2, 3, 4].map((bar) => (
   288→              <motion.div
   289→                key={bar}
   290→                className={`w-[3px] rounded-full ${bar <= barCount ? config.barColor : 'bg-white/20'}`}
   291→                animate={{ height: [4, bar <= barCount ? bar * 3.5 : 4, bar <= barCount ? bar * 3.5 : 4] }}
   292→                transition={{ duration: 0.5, delay: bar * 0.05, ease: 'easeOut' }}
   293→              />
   294→            ))}
   295→          </div>
   296→          {quality === 'poor' ? <Signal size={12} className={config.color} /> : <Wifi size={12} className={config.color} />}
   297→        </div>
   298→      </TooltipTrigger>
   299→      <TooltipContent side="bottom" className="bg-slate-800/95 backdrop-blur-xl text-white border-white/10 text-xs rounded-lg">
   300→        <div className="flex flex-col gap-0.5">
   301→          <span>Network: <span className={config.color}>{config.label}</span></span>
   302→          <span className="text-white/50">Latency: {latency}ms</span>
   303→        </div>
   304→      </TooltipContent>
   305→    </Tooltip>
   306→  );
   307→}
