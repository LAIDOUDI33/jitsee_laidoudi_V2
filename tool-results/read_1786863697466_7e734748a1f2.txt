   264→// ─── Network Quality Indicator ────────────────────────────────
   265→function NetworkQualityIndicator() {
   266→  const [quality, setQuality] = useState<NetworkQuality>('excellent');
   267→  const [latency, setLatency] = useState(22);
   268→
   269→  useEffect(() => {
   270→    const qualities: NetworkQuality[] = ['excellent', 'good', 'fair', 'poor'];
   271→    const changeQuality = () => {
   272→      const q = qualities[Math.floor(Math.random() * qualities.length)];
   273→      setQuality(q);
   274→      const [minLat, maxLat] = networkQualityConfig[q].latency;
   275→      setLatency(Math.floor(Math.random() * (maxLat - minLat + 1)) + minLat);
   276→    };
   277→    const interval = setInterval(changeQuality, 10000 + Math.random() * 5000);
   278→    return () => clearInterval(interval);
   279→  }, []);
   280→
   281→  const config = networkQualityConfig[quality];
   282→  const barCount = quality === 'excellent' ? 4 : quality === 'good' ? 3 : quality === 'fair' ? 2 : 1;
   283→
   284→  return (
   285→    <Tooltip>
   286→      <TooltipTrigger asChild>
   287→        <div className="absolute top-4 right-4 z-30 flex items-center gap-2 px-2.5 py-1.5 rounded-xl bg-black/40 backdrop-blur-xl border border-white/10 cursor-default">
   288→          <div className="flex items-end gap-[2px] h-3.5">
   289→            {[1, 2, 3, 4].map((bar) => (
   290→              <motion.div
   291→                key={bar}
   292→                className={`w-[3px] rounded-full ${bar <= barCount ? config.barColor : 'bg-white/20'}`}
   293→                animate={{ height: [4, bar <= barCount ? bar * 3.5 : 4, bar <= barCount ? bar * 3.5 : 4] }}
   294→                transition={{ duration: 0.5, delay: bar * 0.05, ease: 'easeOut' }}
   295→              />
   296→            ))}
   297→          </div>
   298→          {quality === 'poor' ? <Signal size={12} className={config.color} /> : <Wifi size={12} className={config.color} />}
   299→        </div>
   300→      </TooltipTrigger>
   301→      <TooltipContent side="bottom" className="bg-slate-800/95 backdrop-blur-xl text-white border-white/10 text-xs rounded-lg">
   302→        <div className="flex flex-col gap-0.5">
   303→          <span>Network: <span className={config.color}>{config.label}</span></span>
   304→          <span className="text-white/50">Latency: {latency}ms</span>
   305→        </div>
   306→      </TooltipContent>
   307→    </Tooltip>
   308→  );
   309→}
