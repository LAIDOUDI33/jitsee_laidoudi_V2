   180→function AnimatedCounter({ target }: { target: number }) {
   181→  const [count, setCount] = useState(0)
   182→  const started = useRef(false)
   183→  useEffect(() => {
   184→    if (started.current) return
   185→    started.current = true
   186→    const duration = 1200
   187→    const startTime = performance.now()
   188→    const step = (now: number) => {
   189→      const progress = Math.min((now - startTime) / duration, 1)
   190→      const eased = 1 - Math.pow(1 - progress, 3)
   191→      setCount(Math.round(eased * target))
   192→      if (progress < 1) requestAnimationFrame(step)
   193→    }
   194→    requestAnimationFrame(step)
   195→  }, [target])
   196→  return <span>{count}</span>
   197→}
