    95→    95→function getCountdown(date: string, time: string): string | null {
    96→    96→  if (!date) return null
    97→    97→  const target = new Date(`${date}T${time === '9:00 AM' ? '09:00' : time}`)
    98→    98→  const now = new Date()
    99→    99→  const diff = target.getTime() - now.getTime()
   100→   100→  if (diff <= 0) return null
   101→   101→  const hours = Math.floor(diff / (1000 * 60 * 60))
   102→   102→  const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   103→   103→  if (hours > 24) return `Starts in ${Math.floor(hours / 24)}d ${hours % 24}h`
   104→   104→  if (hours > 0) return `Starts in ${hours}h ${mins}m`
   105→   105→  return `Starts in ${mins}m`
   106→   106→}
