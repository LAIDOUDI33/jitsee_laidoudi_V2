   159→            APP.UI.notifyTokenAuthFailed();

   161→            APP.UI.notifyConnectionFailed(err);
   162→        }
   163→        throw err;
   164→    });

   230→/**
   231→ *
   232→ */
   233→class ConferenceConnector {
   234→    /**
   235→     *
   236→     */
   237→    constructor(resolve, reject) {
   238→        this._resolve = resolve;
   239→        this._reject = reject;
   240→        this.reconnectTimeout = null;
   241→        room.on(JitsiConferenceEvents.CONFERENCE_JOINED,
   242→            this._handleConferenceJoined.bind(this));
   243→        room.on(JitsiConferenceEvents.CONFERENCE_FAILED,
   244→            this._onConferenceFailed.bind(this));
   245→    }
   246→
   247→    /**
   248→     *
   249→     */
   250→    _handleConferenceFailed(err) {
   251→        this._unsubscribe();
   252→        this._reject(err);
   253→    }
   254→
   255→    /**
   256→     *
   257→     */
   258→    _onConferenceFailed(err, ...params) {
   259→        APP.store.dispatch(conferenceFailed(room, err, ...params));
   260→        logger.error('CONFERENCE FAILED:', err, ...params);
   261→
   262→        switch (err) {
   263→        case JitsiConferenceErrors.CONNECTION_ERROR: {
   264→            const [ msg ] = params;
   265→
   266→            APP.UI.notifyConnectionFailed(msg);
   267→            break;
   268→        }
   269→
   270→        case JitsiConferenceErrors.NOT_ALLOWED_ERROR: {
   271→            // let's show some auth not allowed page
   272→            APP.store.dispatch(redirectToStaticPage('static/authError.html'));
   273→            break;
   274→        }
   275→
   276→        // not enough rights to create conference
   277→        case JitsiConferenceErrors.AUTHENTICATION_REQUIRED: {
   278→            // Schedule reconnect to check if someone else created the room.
   279→            this.reconnectTimeout = setTimeout(() => {
   280→                APP.store.dispatch(conferenceWillJoin(room));
   281→                room.join();
   282→            }, 5000);
   283→
   284→            const { password }
   285→                = APP.store.getState()['features/base/conference'];
   286→
   287→            AuthHandler.requireAuth(room, password);
   288→
   289→            break;
   290→        }
   291→
   292→        case JitsiConferenceErrors.RESERVATION_ERROR: {
   293→            const [ code, msg ] = params;
   294→
   295→            APP.UI.notifyReservationError(code, msg);
   296→            break;
   297→        }
   298→
   299→        case JitsiConferenceErrors.GRACEFUL_SHUTDOWN:
   300→            APP.UI.notifyGracefulShutdown();
   301→            break;
   302→
   303→        case JitsiConferenceErrors.CONFERENCE_DESTROYED: {
   304→            const [ reason ] = params;
   305→
   306→            APP.UI.hideStats();
   307→            APP.UI.notifyConferenceDestroyed(reason);
   308→            break;
   309→        }
   310→
   311→        // FIXME FOCUS_DISCONNECTED is a confusing event name.
   312→        // What really happens there is that the library is not ready yet,
   313→        // because Jicofo is not available, but it is going to give it another
   314→        // try.
   315→        case JitsiConferenceErrors.FOCUS_DISCONNECTED: {
   316→            const [ focus, retrySec ] = params;
   317→
   318→            APP.UI.notifyFocusDisconnected(focus, retrySec);
   319→            break;
   320→        }
   321→
   322→        case JitsiConferenceErrors.FOCUS_LEFT:
   323→        case JitsiConferenceErrors.VIDEOBRIDGE_NOT_AVAILABLE:
   324→        case JitsiConferenceErrors.OFFER_ANSWER_FAILED:
   325→            APP.store.dispatch(conferenceWillLeave(room));
   326→
   327→            // FIXME the conference should be stopped by the library and not by
   328→            // the app. Both the errors above are unrecoverable from the library
   329→            // perspective.
   330→            room.leave().then(() => connection.disconnect());
   331→            break;
   332→
   333→        case JitsiConferenceErrors.CONFERENCE_MAX_USERS:
   334→            connection.disconnect();
   335→            APP.UI.notifyMaxUsersLimitReached();
   336→            break;
   337→
   338→        case JitsiConferenceErrors.INCOMPATIBLE_SERVER_VERSIONS:
   339→            APP.store.dispatch(reloadWithStoredParams());
   340→            break;
   341→
   342→        default:
   343→            this._handleConferenceFailed(err, ...params);
   344→        }
   345→    }
   346→
   347→    /**
   348→     *
   349→     */
   350→    _unsubscribe() {
   351→        room.off(
   352→            JitsiConferenceEvents.CONFERENCE_JOINED,
   353→            this._handleConferenceJoined);
   354→        room.off(
   355→            JitsiConferenceEvents.CONFERENCE_FAILED,
   356→            this._onConferenceFailed);
   357→        if (this.reconnectTimeout !== null) {
   358→            clearTimeout(this.reconnectTimeout);
   359→        }
   360→        AuthHandler.closeAuth();
   361→    }
   362→
   363→    /**
   364→     *
   365→     */
   366→    _handleConferenceJoined() {
   367→        this._unsubscribe();
   368→        this._resolve();
   369→    }
   370→
   371→    /**
   372→     *
   373→     */
   374→    connect() {
   375→        room.join();
   376→    }
   377→}
   378→
   379→/**
   380→ * Disconnects the connection.
   381→ * @returns resolved Promise. We need this in order to make the Promise.all
   382→ * call in hangup() to resolve when all operations are finished.
   383→ */
   384→function disconnect() {
   385→    const onDisconnected = () => {
   386→        APP.API.notifyConferenceLeft(APP.conference.roomName);
   387→
   388→        return Promise.resolve();
   389→    };
   390→
   391→    return connection.disconnect().then(onDisconnected, onDisconnected);
   392→}
   393→
   394→/**
   395→ * Handles CONNECTION_FAILED events from lib-jitsi-meet.
   396→ *
   397→ * @param {JitsiConnectionError} error - The reported error.
   398→ * @returns {void}
   399→ * @private
   400→ */
   401→function _connectionFailedHandler(error) {
   402→    if (isFatalJitsiConnectionError(error)) {
   403→        APP.connection.removeEventListener(
   404→            JitsiConnectionEvents.CONNECTION_FAILED,
   405→            _connectionFailedHandler);
   406→        if (room) {
   407→            APP.store.dispatch(conferenceWillLeave(room));
   408→            room.leave();
   409→        }
   410→    }
   411→}

   413→export default {
   414→    /**
   415→     * Flag used to delay modification of the muted status of local media tracks
   416→     * until those are created (or not, but at that point it's certain that
   417→     * the tracks won't exist).
   418→     */
   419→    _localTracksInitialized: false,
   420→    isModerator: false,
   421→    isSharingScreen: false,
   422→
   423→    /**
   424→     * Indicates if the desktop sharing functionality has been enabled.
   425→     * It takes into consideration the status returned by
   426→     * {@link JitsiMeetJS.isDesktopSharingEnabled()}. The latter can be false
   427→     * either if the desktop sharing is not supported by the current browser
   428→     * or if it was disabled through lib-jitsi-meet specific options (check
   429→     * config.js for listed options).
   430→     */
   431→    isDesktopSharingEnabled: false,
   432→
   433→    /**
   434→     * The local audio track (if any).
   435→     * FIXME tracks from redux store should be the single source of truth
   436→     * @type {JitsiLocalTrack|null}
   437→     */
   438→    localAudio: null,
   439→
   440→    /**
   441→     * The local video track (if any).
   442→     * FIXME tracks from redux store should be the single source of truth, but
   443→     * more refactoring is required around screen sharing ('localVideo' usages).
   444→     * @type {JitsiLocalTrack|null}
   445→     */
   446→    localVideo: null,
   447→
   448→    /**
   449→     * Creates local media tracks and connects to a room. Will show error
   450→     * dialogs in case accessing the local microphone and/or camera failed. Will
   451→     * show guidance overlay for users on how to give access to camera and/or
   452→     * microphone.
   453→     * @param {string} roomName
   454→     * @param {object} options
   455→     * @param {boolean} options.startAudioOnly=false - if <tt>true</tt> then
   456→     * only audio track will be created and the audio only mode will be turned
   457→     * on.
   458→     * @param {boolean} options.startScreenSharing=false - if <tt>true</tt>
   459→     * should start with screensharing instead of camera video.
   460→     * @param {boolean} options.startWithAudioMuted - will start the conference
   461→     * without any audio tracks.
   462→     * @param {boolean} options.startWithVideoMuted - will start the conference
   463→     * without any video tracks.
   464→     * @returns {Promise.<JitsiLocalTrack[], JitsiConnection>}
   465→     */
   466→    createInitialLocalTracksAndConnect(roomName, options = {}) {
   467→        let audioAndVideoError,
   468→            audioOnlyError,
   469→            screenSharingError,
   470→            videoOnlyError;
   471→        const initialDevices = [];
   472→        let requestedAudio = false;
   473→        let requestedVideo = false;
   474→
   475→        if (!options.startWithAudioMuted) {
   476→            initialDevices.push('audio');
   477→            requestedAudio = true;
   478→        }
   479→        if (!options.startWithVideoMuted
   480→                && !options.startAudioOnly
   481→                && !options.startScreenSharing) {
   482→            initialDevices.push('video');
   483→            requestedVideo = true;
   484→        }
   485→
   486→        JitsiMeetJS.mediaDevices.addEventListener(
   487→            JitsiMediaDevicesEvents.PERMISSION_PROMPT_IS_SHOWN,
   488→            browser =>
   489→                APP.store.dispatch(
   490→                    mediaPermissionPromptVisibilityChanged(true, browser))
   491→        );
   492→
   493→        let tryCreateLocalTracks;
   494→
   495→        // FIXME is there any simpler way to rewrite this spaghetti below ?
   496→        if (options.startScreenSharing) {
   497→            tryCreateLocalTracks = this._createDesktopTrack()
   498→                .then(desktopStream => {
   499→                    if (!requestedAudio) {
   500→                        return [ desktopStream ];
   501→                    }
   502→
   503→                    return createLocalTracksF({ devices: [ 'audio' ] }, true)
   504→                        .then(([ audioStream ]) =>
   505→                            [ desktopStream, audioStream ])
   506→                        .catch(error => {
   507→                            audioOnlyError = error;
   508→
   509→                            return [ desktopStream ];
   510→                        });
   511→                })
   512→                .catch(error => {
   513→                    logger.error('Failed to obtain desktop stream', error);
   514→                    screenSharingError = error;
   515→
   516→                    return requestedAudio
   517→                        ? createLocalTracksF({ devices: [ 'audio' ] }, true)
   518→                        : [];
   519→                })
   520→                .catch(error => {
   521→                    audioOnlyError = error;
   522→
   523→                    return [];
   524→                });
   525→        } else if (!requestedAudio && !requestedVideo) {
   526→            // Resolve with no tracks
   527→            tryCreateLocalTracks = Promise.resolve([]);
   528→        } else {
   529→            tryCreateLocalTracks = createLocalTracksF({ devices: initialDevices }, true)
   530→                .catch(err => {
   531→                    if (requestedAudio && requestedVideo) {
   532→
   533→                        // Try audio only...
   534→                        audioAndVideoError = err;
   535→
   536→                        return (
   537→                            createLocalTracksF({ devices: [ 'audio' ] }, true));
   538→                    } else if (requestedAudio && !requestedVideo) {
   539→                        audioOnlyError = err;
   540→
   541→                        return [];
   542→                    } else if (requestedVideo && !requestedAudio) {
   543→                        videoOnlyError = err;
   544→
   545→                        return [];
   546→                    }
   547→                    logger.error('Should never happen');
   548→                })
   549→                .catch(err => {
   550→                    // Log this just in case...
   551→                    if (!requestedAudio) {
   552→                        logger.error('The impossible just happened', err);
   553→                    }
   554→                    audioOnlyError = err;
   555→
   556→                    // Try video only...
   557→                    return requestedVideo
   558→                        ? createLocalTracksF({ devices: [ 'video' ] }, true)
   559→                        : [];
   560→                })
   561→                .catch(err => {
   562→                    // Log this just in case...
   563→                    if (!requestedVideo) {
   564→                        logger.error('The impossible just happened', err);
   565→                    }
   566→                    videoOnlyError = err;
   567→
   568→                    return [];
   569→                });
   570→        }
   571→
   572→        // Hide the permissions prompt/overlay as soon as the tracks are
   573→        // created. Don't wait for the connection to be made, since in some
   574→        // cases, when auth is rquired, for instance, that won't happen until
   575→        // the user inputs their credentials, but the dialog would be
   576→        // overshadowed by the overlay.
   577→        tryCreateLocalTracks.then(() =>
   578→            APP.store.dispatch(mediaPermissionPromptVisibilityChanged(false)));
   579→
   580→        return Promise.all([ tryCreateLocalTracks, connect(roomName) ])
   581→            .then(([ tracks, con ]) => {
   582→                // FIXME If there will be microphone error it will cover any
   583→                // screensharing dialog, but it's still better than in
   584→                // the reverse order where the screensharing dialog will
   585→                // sometimes be closing the microphone alert ($.prompt.close();
   586→                // is called). Need to figure out dialogs chaining to fix that.
   587→                if (screenSharingError) {
   588→                    this._handleScreenSharingError(screenSharingError);
   589→                }
   590→                if (audioAndVideoError || audioOnlyError) {
   591→                    if (audioOnlyError || videoOnlyError) {
   592→                        // If both requests for 'audio' + 'video' and 'audio'
   593→                        // only failed, we assume that there are some problems
   594→                        // with user's microphone and show corresponding dialog.
   595→                        APP.store.dispatch(notifyMicError(audioOnlyError));
   596→                        APP.store.dispatch(notifyCameraError(videoOnlyError));
   597→                    } else {
   598→                        // If request for 'audio' + 'video' failed, but request
   599→                        // for 'audio' only was OK, we assume that we had
   600→                        // problems with camera and show corresponding dialog.
   601→                        APP.store.dispatch(
   602→                            notifyCameraError(audioAndVideoError));
   603→                    }
   604→                }
   605→
   606→                return [ tracks, con ];
   607→            });
   608→    },
   609→
   610→    /**
   611→     * Open new connection and join to the conference.
   612→     * @param {object} options
   613→     * @param {string} roomName - The name of the conference.
   614→     * @returns {Promise}
   615→     */
   616→    init(options) {
   617→        this.roomName = options.roomName;
   618→
   619→        return (
   620→
   621→            // Initialize the device list first. This way, when creating tracks
   622→            // based on preferred devices, loose label matching can be done in
   623→            // cases where the exact ID match is no longer available, such as
   624→            // when the camera device has switched USB ports.
   625→            // when in startSilent mode we want to start with audio muted
   626→            this._initDeviceList()
   627→                .catch(error => logger.warn(
   628→                    'initial device list initialization failed', error))
   629→                .then(() => this.createInitialLocalTracksAndConnect(
   630→                options.roomName, {
   631→                    startAudioOnly: config.startAudioOnly,
   632→                    startScreenSharing: config.startScreenSharing,
   633→                    startWithAudioMuted: config.startWithAudioMuted
   634→                        || config.startSilent
   635→                        || isUserInteractionRequiredForUnmute(APP.store.getState()),
   636→                    startWithVideoMuted: config.startWithVideoMuted
   637→                        || isUserInteractionRequiredForUnmute(APP.store.getState())
   638→                }))
   639→            .then(([ tracks, con ]) => {
   640→                tracks.forEach(track => {
   641→                    if ((track.isAudioTrack() && this.isLocalAudioMuted())
   642→                        || (track.isVideoTrack() && this.isLocalVideoMuted())) {
   643→                        const mediaType = track.getType();
   644→
   645→                        sendAnalytics(
   646→                            createTrackMutedEvent(mediaType, 'initial mute'));
   647→                        logger.log(`${mediaType} mute: initially muted.`);
   648→                        track.mute();
   649→                    }
   650→                });
   651→                logger.log(`initialized with ${tracks.length} local tracks`);
   652→                this._localTracksInitialized = true;
   653→                con.addEventListener(
   654→                    JitsiConnectionEvents.CONNECTION_FAILED,
   655→                    _connectionFailedHandler);
   656→                APP.connection = connection = con;

  1291→    /**
  1292→     * Start using provided audio stream.
  1293→     * Stops previous audio stream.
  1294→     * @param {JitsiLocalTrack} [stream] new stream to use or null
  1295→     * @returns {Promise}

  1304→                        this.setAudioMuteStatus(this.isLocalAudioMuted());
  1305→                    })

  1313→    /**
  1314→     * Returns whether or not the conference is currently in audio only mode.

  1320→    },
  1321→
  1322→    videoSwitchInProgress: false,
  1323→
  1324→    /**
  1325→     * This fields stores a handler which will create a Promise which turns off
  1326→     * the screen sharing and restores the previous video state (was there
  1327→     * any video, before switching to screen sharing ? was it muted ?).
  1328→     *
  1329→     * Once called this fields is cleared to <tt>null</tt>.
  1330→     * @type {Function|null}
  1331→     */
  1332→    _untoggleScreenSharing: null,
  1333→
  1334→    /**
  1335→     * Creates a Promise which turns off the screen sharing and restores
  1336→     * the previous state described by the arguments.
  1337→     *
  1338→     * This method is bound to the appropriate values, after switching to screen
  1339→     * sharing and stored in {@link _untoggleScreenSharing}.
  1340→     *
  1341→     * @param {boolean} didHaveVideo indicates if there was a camera video being
  1342→     * used, before switching to screen sharing.
  1343→     * @param {boolean} wasVideoMuted indicates if the video was muted, before
  1344→     * switching to screen sharing.
  1345→     * @return {Promise} resolved after the screen sharing is turned off, or
  1346→     * rejected with some error (no idea what kind of error, possible GUM error)
  1347→     * in case it fails.
  1348→     * @private
  1349→     */
  1350→    _turnScreenSharingOff(didHaveVideo, wasVideoMuted) {
  1351→        this._untoggleScreenSharing = null;
  1352→        this.videoSwitchInProgress = true;
  1353→        const { receiver } = APP.remoteControl;
  1354→
  1355→        if (receiver) {
  1356→            receiver.stop();
  1357→        }
  1358→
  1359→        this._stopProxyConnection();
  1360→
  1361→        let promise = null;
  1362→
  1363→        if (didHaveVideo) {
  1364→            promise = createLocalTracksF({ devices: [ 'video' ] })
  1365→                .then(([ stream ]) => this.useVideoStream(stream))
  1366→                .then(() => {
  1367→                    sendAnalytics(createScreenSharingEvent('stopped'));
  1368→                    logger.log('Screen sharing stopped, switching to video.');
  1369→
  1370→                    if (!this.localVideo && wasVideoMuted) {
  1371→                        return Promise.reject('No local video to be muted!');
  1372→                    } else if (wasVideoMuted && this.localVideo) {
  1373→                        return this.localVideo.mute();
  1374→                    }
  1375→                })
  1376→                .catch(error => {
  1377→                    logger.error('failed to switch back to local video', error);
  1378→
  1379→                    return this.useVideoStream(null).then(() =>
  1380→
  1381→                        // Still fail with the original err
  1382→                        Promise.reject(error)
  1383→                    );
  1384→                });
  1385→        } else {
  1386→            promise = this.useVideoStream(null);
  1387→        }
  1388→
  1389→        return promise.then(
  1390→            () => {
  1391→                this.videoSwitchInProgress = false;
  1392→            },
  1393→            error => {
  1394→                this.videoSwitchInProgress = false;
  1395→                throw error;
  1396→            });
  1397→    },
  1398→
  1399→    /**
  1400→     * Toggles between screen sharing and camera video if the toggle parameter
  1401→     * is not specified and starts the procedure for obtaining new screen
  1402→     * sharing/video track otherwise.
  1403→     *
  1404→     * @param {boolean} [toggle] - If true - new screen sharing track will be
  1405→     * obtained. If false - new video track will be obtain. If not specified -
  1406→     * toggles between screen sharing and camera video.
  1407→     * @param {Object} [options] - Screen sharing options that will be passed to
  1408→     * createLocalTracks.
  1409→     * @param {Array<string>} [options.desktopSharingSources] - Array with the
  1410→     * sources that have to be displayed in the desktop picker window ('screen',
  1411→     * 'window', etc.).
  1412→     * @return {Promise.<T>}
  1413→     */
  1414→    toggleScreenSharing(toggle = !this._untoggleScreenSharing, options = {}) {
  1415→        if (this.videoSwitchInProgress) {
  1416→            return Promise.reject('Switch in progress.');
  1417→        }
  1418→        if (!this.isDesktopSharingEnabled) {
  1419→            return Promise.reject(
  1420→                'Cannot toggle screen sharing: not supported.');
  1421→        }
  1422→
  1423→        if (this.isAudioOnly()) {
  1424→            return Promise.reject('No screensharing in audio only mode');
  1425→        }
  1426→
  1427→        if (toggle) {
  1428→            return this._switchToScreenSharing(options);
  1429→        }
  1430→
  1431→        return this._untoggleScreenSharing
  1432→            ? this._untoggleScreenSharing()
  1433→            : Promise.resolve();
  1434→    },
  1435→
  1436→    /**
  1437→     * Creates desktop (screensharing) {@link JitsiLocalTrack}
  1438→     *
  1439→     * @param {Object} [options] - Screen sharing options that will be passed to
  1440→     * createLocalTracks.
  1441→     * @param {Object} [options.desktopSharing]
  1442→     * @param {Object} [options.desktopStream] - An existing desktop stream to
  1443→     * use instead of creating a new desktop stream.
  1444→     * @return {Promise.<JitsiLocalTrack>} - A Promise resolved with
  1445→     * {@link JitsiLocalTrack} for the screensharing or rejected with
  1446→     * {@link JitsiTrackError}.
  1447→     *
  1448→     * @private
  1449→     */
  1450→    _createDesktopTrack(options = {}) {
  1451→        let externalInstallation = false;
  1452→        let DSExternalInstallationInProgress = false;
  1453→        const didHaveVideo = Boolean(this.localVideo);
  1454→        const wasVideoMuted = this.isLocalVideoMuted();
  1455→
  1456→        const getDesktopStreamPromise = options.desktopStream
  1457→            ? Promise.resolve([ options.desktopStream ])
  1458→            : createLocalTracksF({
  1459→                desktopSharingSourceDevice: options.desktopSharingSources
  1460→                    ? null : config._desktopSharingSourceDevice,
  1461→                desktopSharingSources: options.desktopSharingSources,
  1462→                devices: [ 'desktop' ],
  1463→                desktopSharingExtensionExternalInstallation: {
  1464→                    interval: 500,
  1465→                    checkAgain: () => DSExternalInstallationInProgress,
  1466→                    listener: (status, url) => {
  1467→                        switch (status) {
  1468→                        case 'waitingForExtension': {
  1469→                            DSExternalInstallationInProgress = true;
  1470→                            externalInstallation = true;
  1471→                            const listener = () => {
  1472→                                // Wait a little bit more just to be sure that
  1473→                                // we won't miss the extension installation
  1474→                                setTimeout(() => {
  1475→                                    DSExternalInstallationInProgress = false;
  1476→                                },
  1477→                                500);
  1478→                                APP.UI.removeListener(
  1479→                                    UIEvents.EXTERNAL_INSTALLATION_CANCELED,
  1480→                                    listener);
  1481→                            };
  1482→
  1483→                            APP.UI.addListener(
  1484→                                UIEvents.EXTERNAL_INSTALLATION_CANCELED,
  1485→                                listener);
  1486→                            APP.UI.showExtensionExternalInstallationDialog(url);
  1487→                            break;
  1488→                        }
  1489→                        case 'extensionFound':
  1490→                            // Close the dialog.
  1491→                            externalInstallation && $.prompt.close();
  1492→                            break;
  1493→                        default:
  1494→
  1495→                            // Unknown status
  1496→                        }
  1497→                    }
  1498→                }
  1499→            });
  1500→
  1501→        return getDesktopStreamPromise.then(([ desktopStream ]) => {
  1502→            // Stores the "untoggle" handler which remembers whether was
  1503→            // there any video before and whether was it muted.
  1504→            this._untoggleScreenSharing
  1505→                = this._turnScreenSharingOff
  1506→                      .bind(this, didHaveVideo, wasVideoMuted);
  1507→            desktopStream.on(
  1508→                JitsiTrackEvents.LOCAL_TRACK_STOPPED,
  1509→                () => {
  1510→                    // If the stream was stopped during screen sharing
  1511→                    // session then we should switch back to video.
  1512→                    this.isSharingScreen
  1513→                        && this._untoggleScreenSharing
  1514→                        && this._untoggleScreenSharing();
  1515→                }
  1516→            );
  1517→
  1518→            // close external installation dialog on success.
  1519→            externalInstallation && $.prompt.close();
  1520→
  1521→            return desktopStream;
  1522→        }, error => {
  1523→            DSExternalInstallationInProgress = false;
  1524→
  1525→            // close external installation dialog on success.
  1526→            externalInstallation && $.prompt.close();
  1527→            throw error;
  1528→        });
  1529→    },
  1530→
  1531→    /**
  1532→     * Tries to switch to the screensharing mode by disposing camera stream and
  1533→     * replacing it with a desktop one.
  1534→     *
  1535→     * @param {Object} [options] - Screen sharing options that will be passed to
  1536→     * createLocalTracks.
  1537→     *
  1538→     * @return {Promise} - A Promise resolved if the operation succeeds or
  1539→     * rejected with some unknown type of error in case it fails. Promise will
  1540→     * be rejected immediately if {@link videoSwitchInProgress} is true.
  1541→     *
  1542→     * @private
  1543→     */
  1544→    _switchToScreenSharing(options = {}) {
  1545→        if (this.videoSwitchInProgress) {
  1546→            return Promise.reject('Switch in progress.');
  1547→        }
  1548→
  1549→        this.videoSwitchInProgress = true;
  1550→
  1551→        return this._createDesktopTrack(options)
  1552→            .then(stream => this.useVideoStream(stream))
  1553→            .then(() => {
  1554→                this.videoSwitchInProgress = false;
  1555→                sendAnalytics(createScreenSharingEvent('started'));
  1556→                logger.log('Screen sharing started');
  1557→            })
  1558→            .catch(error => {
  1559→                this.videoSwitchInProgress = false;
  1560→
  1561→                // Pawel: With this call I'm trying to preserve the original
  1562→                // behaviour although it is not clear why would we "untoggle"
  1563→                // on failure. I suppose it was to restore video in case there
  1564→                // was some problem during "this.useVideoStream(desktopStream)".
  1565→                // It's important to note that the handler will not be available
  1566→                // if we fail early on trying to get desktop media (which makes
  1567→                // sense, because the camera video is still being used, so
  1568→                // nothing to "untoggle").
  1569→                if (this._untoggleScreenSharing) {
  1570→                    this._untoggleScreenSharing();
  1571→                }
  1572→
  1573→                // FIXME the code inside of _handleScreenSharingError is
  1574→                // asynchronous, but does not return a Promise and is not part
  1575→                // of the current Promise chain.
  1576→                this._handleScreenSharingError(error);
  1577→
  1578→                return Promise.reject(error);
  1579→            });
  1580→    },
  1581→
  1582→    /**
  1583→     * Handles {@link JitsiTrackError} returned by the lib-jitsi-meet when
  1584→     * trying to create screensharing track. It will either do nothing if
  1585→     * the dialog was canceled on user's request or display inline installation
  1586→     * dialog and ask the user to install the extension, once the extension is
  1587→     * installed it will switch the conference to screensharing. The last option
  1588→     * is that an unrecoverable error dialog will be displayed.
  1589→     * @param {JitsiTrackError} error - The error returned by
  1590→     * {@link _createDesktopTrack} Promise.
  1591→     * @private
  1592→     */
  1593→    _handleScreenSharingError(error) {
  1594→        if (error.name === JitsiTrackErrors.CHROME_EXTENSION_USER_CANCELED) {
  1595→            return;
  1596→        }
  1597→
  1598→        logger.error('failed to share local desktop', error);
  1599→
  1600→        if (error.name
  1601→                === JitsiTrackErrors.CHROME_EXTENSION_USER_GESTURE_REQUIRED) {
  1602→            // If start with screen sharing the extension will fail to install
  1603→            // (if not found), because the request has been triggered by the
  1604→            // script. Show a dialog which asks user to click "install" and try
  1605→            // again switching to the screen sharing.
  1606→            APP.UI.showExtensionInlineInstallationDialog(
  1607→                () => {
  1608→                    // eslint-disable-next-line no-empty-function
  1609→                    this.toggleScreenSharing().catch(() => {});
  1610→                }
  1611→            );
  1612→
  1613→            return;
  1614→        }
  1615→
  1616→        // Handling:
  1617→        // JitsiTrackErrors.PERMISSION_DENIED
  1618→        // JitsiTrackErrors.CHROME_EXTENSION_INSTALLATION_ERROR
  1619→        // JitsiTrackErrors.CONSTRAINT_FAILED
  1620→        // JitsiTrackErrors.GENERAL
  1621→        // and any other
  1622→        let descriptionKey;
  1623→        let titleKey;
  1624→
  1625→        if (error.name === JitsiTrackErrors.PERMISSION_DENIED) {
  1626→
  1627→            // in FF the only option for user is to deny access temporary or
  1628→            // permanently and we only receive permission_denied
  1629→            // we always show some info cause in case of permanently, no info
  1630→            // shown will be bad experience
  1631→            //
  1632→            // TODO: detect interval between requesting permissions and received
  1633→            // error, this way we can detect user interaction which will have
  1634→            // longer delay
  1635→            if (JitsiMeetJS.util.browser.isFirefox()) {
  1636→                descriptionKey
  1637→                    = 'dialog.screenSharingFirefoxPermissionDeniedError';
  1638→                titleKey = 'dialog.screenSharingFirefoxPermissionDeniedTitle';
  1639→            } else {
  1640→                descriptionKey = 'dialog.screenSharingPermissionDeniedError';
  1641→                titleKey = 'dialog.screenSharingFailedToInstallTitle';
  1642→            }
  1643→        } else if (error.name === JitsiTrackErrors.CONSTRAINT_FAILED) {
  1644→            descriptionKey = 'dialog.cameraConstraintFailedError';
  1645→            titleKey = 'deviceError.cameraError';
  1646→        } else {
  1647→            descriptionKey = 'dialog.screenSharingFailedToInstall';
  1648→            titleKey = 'dialog.screenSharingFailedToInstallTitle';
  1649→        }
  1650→
  1651→        APP.UI.messageHandler.showError({
  1652→            descriptionKey,
  1653→            titleKey
  1654→        });
  1655→    },
