   320→function generateRoomId(): string {
   321→  const group = () =>
   322→    Array.from({ length: 3 }, () =>
   323→      String.fromCharCode(97 + Math.floor(Math.random() * 26)),
   324→    ).join('');
   325→  return `${group()}-${group()}-${group()}`;
   326→}

   328→/* -------------------------------------------------------------------------- */
   329→/*                          ANIMATED COUNTER COMPONENT                       */
   330→/* -------------------------------------------------------------------------- */
   331→
   332→function AnimatedCounter({ target, suffix, label }: StatItem) {
   333→  const ref = useRef<HTMLSpanElement>(null);
   334→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   335→  const [count, setCount] = useState(0);
   336→
   337→  useEffect(() => {
   338→    if (!isInView) return;
   339→
   340→    const duration = 2000;
   341→    const steps = 60;
   342→    const increment = target / steps;
   343→    let current = 0;
   344→    const timer = setInterval(() => {
   345→      current += increment;
   346→      if (current >= target) {
   347→        setCount(target);
   348→        clearInterval(timer);
   349→      } else {
   350→        setCount(current);
   351→      }
   352→    }, duration / steps);
   353→
   354→    return () => clearInterval(timer);
   355→  }, [isInView, target]);
   356→
   357→  const displayValue =
   358→    target % 1 !== 0 ? count.toFixed(1) : Math.round(count).toLocaleString();
   359→
   360→  return (
   361→    <div className="text-center">
   362→      <div className="text-4xl md:text-5xl font-bold tracking-tight">
   363→        <span ref={ref}>{displayValue}</span>
   364→        <span className="gradient-text">{suffix}</span>
   365→      </div>
   366→      <p className="mt-2 text-muted-foreground text-sm font-medium">{label}</p>
   367→    </div>
   368→  );
   369→}
