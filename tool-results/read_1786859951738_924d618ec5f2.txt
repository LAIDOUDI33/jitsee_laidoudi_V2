   309→function generateRoomId(): string {
   310→  const group = () =>
   311→    Array.from({ length: 3 }, () =>
   312→      String.fromCharCode(97 + Math.floor(Math.random() * 26)),
   313→    ).join('');
   314→  return `${group()}-${group()}-${group()}`;
   315→}

   317→/* -------------------------------------------------------------------------- */
   318→/*                          ANIMATED COUNTER COMPONENT                       */
   319→/* -------------------------------------------------------------------------- */
   320→
   321→function AnimatedCounter({ target, suffix, label }: StatItem) {
   322→  const ref = useRef<HTMLSpanElement>(null);
   323→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   324→  const [count, setCount] = useState(0);
   325→
   326→  useEffect(() => {
   327→    if (!isInView) return;
   328→
   329→    const duration = 2000;
   330→    const steps = 60;
   331→    const increment = target / steps;
   332→    let current = 0;
   333→    const timer = setInterval(() => {
   334→      current += increment;
   335→      if (current >= target) {
   336→        setCount(target);
   337→        clearInterval(timer);
   338→      } else {
   339→        setCount(current);
   340→      }
   341→    }, duration / steps);
   342→
   343→    return () => clearInterval(timer);
   344→  }, [isInView, target]);
   345→
   346→  const displayValue =
   347→    target % 1 !== 0 ? count.toFixed(1) : Math.round(count).toLocaleString();
   348→
   349→  return (
   350→    <div className="text-center">
   351→      <div className="text-4xl md:text-5xl font-bold tracking-tight">
   352→        <span ref={ref}>{displayValue}</span>
   353→        <span className="gradient-text">{suffix}</span>
   354→      </div>
   355→      <p className="mt-2 text-muted-foreground text-sm font-medium">{label}</p>
   356→    </div>
   357→  );
   358→}
