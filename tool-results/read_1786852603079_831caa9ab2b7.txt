    18→const registerSchema = z.object({

    24→}).refine((data) => data.password === data.confirmPassword, {
    25→  message: 'Passwords do not match',
    26→  path: ['confirmPassword'],
    27→});

    29→type RegisterForm = z.infer<typeof registerSchema>;

    37→function getPasswordStrength(password: string) {
    38→  let score = 0;
    39→  const checks: { label: string; met: boolean }[] = [];
    40→  if (password.length >= 8) { score++; checks.push({ label: '8+ characters', met: true }); } else { checks.push({ label: '8+ characters', met: false }); }
    41→  if (/[A-Z]/.test(password)) { score++; checks.push({ label: 'Uppercase letter', met: true }); } else { checks.push({ label: 'Uppercase letter', met: false }); }
    42→  if (/[0-9]/.test(password)) { score++; checks.push({ label: 'Number', met: true }); } else { checks.push({ label: 'Number', met: false }); }
    43→  if (/[^A-Za-z0-9]/.test(password)) { score++; checks.push({ label: 'Special character', met: true }); } else { checks.push({ label: 'Special character', met: false }); }
    44→
    45→  const config = score <= 1
    46→    ? { label: 'Weak — easily guessed', color: 'bg-red-500', textColor: 'text-red-500', trackColor: 'bg-red-500' }
    47→    : score <= 2
    48→    ? { label: 'Fair — could be stronger', color: 'bg-yellow-500', textColor: 'text-yellow-600', trackColor: 'bg-yellow-500' }
    49→    : score <= 3
    50→    ? { label: 'Good — reasonably strong', color: 'bg-emerald-400', textColor: 'text-emerald-500', trackColor: 'bg-emerald-400' }
    51→    : { label: 'Strong — excellent security', color: 'bg-emerald-500', textColor: 'text-emerald-600', trackColor: 'bg-emerald-500' };
    52→
    53→  return { score, ...config, checks };
    54→}
