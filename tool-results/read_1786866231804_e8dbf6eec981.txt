   139→function TeamSparkline({ data, color }: { data: number[]; color: string }) {
   140→  const maxVal = Math.max(...data)
   141→  return (
   142→    <div className='flex items-end gap-[2px] h-6' aria-label='Team activity last 7 days'>
   143→      {data.map((val, i) => (
   144→        <motion.div
   145→          key={i}
   146→          className={`w-[4px] rounded-sm ${color}`}
   147→          initial={{ height: 0 }}
   148→          animate={{ height: `${(val / maxVal) * 100}%` }}
   149→          transition={{ delay: i * 0.05, duration: 0.3, ease: 'easeOut' }}
   150→          title={`Day ${i + 1}: ${val}%`}
   151→        />
   152→      ))}

   154→  )

   176→function CreateTeamStepIndicator({ step }: { step: number }) {
   177→  const steps = ['Details', 'Members', 'Settings']
   178→  return (
   179→    <div className='flex items-center gap-2 mb-4'>
   180→      {steps.map((label, i) => (
   181→        <div key={label} className='flex items-center gap-2 flex-1'>
   182→          <motion.div
   183→            className={`flex items-center justify-center w-6 h-6 rounded-full text-[10px] font-semibold shrink-0 transition-colors duration-300 ${
   184→              i < step
   185→                ? 'bg-primary text-primary-foreground'
   186→                : i === step
   187→                  ? 'bg-primary/15 text-primary border-2 border-primary'
   188→                  : 'bg-muted text-muted-foreground border-2 border-muted'
   189→            }`}
   190→            animate={i === step ? { scale: [1, 1.1, 1] } : { scale: 1 }}
   191→            transition={{ duration: 0.3 }}
   192→          >
   193→            {i < step ? <Check className='h-3 w-3' /> : i + 1}
   194→          </motion.div>
   195→          <span className={`text-[11px] font-medium truncate transition-colors duration-300 ${
   196→            i <= step ? 'text-foreground' : 'text-muted-foreground'
   197→          }`}>
   198→            {label}
   199→          </span>
   200→          {i < steps.length - 1 && (
   201→            <div className={`flex-1 h-[2px] rounded-full transition-colors duration-300 ${
   202→              i < step ? 'bg-primary' : 'bg-muted'
   203→            }`} />
   204→          )}
   205→        </div>
   206→      ))}
   207→    </div>
   208→  )
