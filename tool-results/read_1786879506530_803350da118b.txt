   144→function StatusIcon({ status }: { status?: string }) {
   145→  if (status === 'read') return <CheckCheck className='h-3.5 w-3.5 text-primary' />
   146→  if (status === 'delivered') return <CheckCheck className='h-3.5 w-3.5 text-muted-foreground' />
   147→  if (status === 'sent') return <Check className='h-3.5 w-3.5 text-muted-foreground' />
   148→  return null
   149→}
