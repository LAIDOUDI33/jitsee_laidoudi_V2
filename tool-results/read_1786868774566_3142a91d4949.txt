   174→function getInitials(name: string): string {
   175→  return name
   176→    .split(' ')
   177→    .map((n) => n[0])
   178→    .join('')
   179→    .slice(0, 2)
   180→    .toUpperCase();
   181→}
   182→
   183→function formatTime(seconds: number): string {
   184→  const m = Math.floor(seconds / 60);
   185→  const s = seconds % 60;
   186→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
   187→}
   188→
   189→function formatTimeLimit(minutes: number): string {
   190→  return formatTime(minutes * 60);
   191→}

   232→function AvatarStack({
   233→  participantIds,
   234→  max = 4,
   235→}: {
   236→  participantIds: string[];
   237→  max?: number;
   238→}) {
   239→  const allParticipants = useParticipants();
   240→  const visible = participantIds.slice(0, max);
   241→  const remaining = participantIds.length - max;
   242→
   243→  return (
   244→    <div className="flex items-center">
   245→      {visible.map((pid, i) => {
   246→        const p = allParticipants.find((x) => x.id === pid);
   247→        if (!p) return null;
   248→        return (
   249→          <TooltipProvider key={pid} delayDuration={0}>
   250→            <Tooltip>
   251→              <TooltipTrigger asChild>
   252→                <div
   253→                  className="h-7 w-7 rounded-full flex items-center justify-center text-[11px] font-bold text-white ring-2 ring-card -ml-[6px] first:ml-0 transition-transform hover:scale-110 hover:z-10 cursor-pointer"
   254→                  style={{ backgroundColor: p.color, zIndex: visible.length - i }}
   255→                >
   256→                  {getInitials(p.name)}
   257→                </div>
   258→              </TooltipTrigger>
   259→              <TooltipContent side="bottom" className="text-xs">
   260→                {p.name}
   261→              </TooltipContent>
   262→            </Tooltip>
   263→          </TooltipProvider>
   264→        );
   265→      })}
   266→      {remaining > 0 && (
   267→        <div className="h-7 w-7 rounded-full flex items-center justify-center text-[10px] font-semibold bg-muted text-muted-foreground ring-2 ring-card -ml-[6px]">
   268→          +{remaining}
   269→        </div>
   270→      )}
   271→    </div>
   272→  );
   273→}

   442→import { createContext, useContext } from 'react';
   443→
   444→interface BreakoutContextType {
   445→  participants: Participant[];
   446→  rooms: BreakoutRoom[];
   447→  unassignedIds: string[];
   448→  settings: RoomSettings;
   449→  getParticipant: (id: string) => Participant | undefined;
   450→}
   451→
   452→const BreakoutContext = createContext<BreakoutContextType | null>(null);
   453→
   454→function useBreakout() {
   455→  const ctx = useContext(BreakoutContext);
   456→  if (!ctx) throw new Error('useBreakout must be used within BreakoutRoomsPage');
   457→  return ctx;
   458→}
   459→
   460→function useParticipants() {
   461→  return useBreakout().participants;
   462→}
   463→
   464→function useRoomSettings() {
   465→  return useBreakout().settings;
   466→}

   472→function UnassignedPool({

   474→  onAssignClick,
   475→}: {
   476→  unassignedIds: string[];
   477→  onAssignClick: (pid: string) => void;
   478→}) {
   479→  const { getParticipant } = useBreakout();

   493→          Unassigned ({unassignedIds.length})
   494→        </span>
   495→      </div>
   496→      <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-thin">
   497→        <AnimatePresence>
   498→          {unassignedIds.map((pid) => {
   499→            const p = getParticipant(pid);
   500→            if (!p) return null;

   502→              <motion.button
   503→                key={pid}
   504→                initial={{ opacity: 0, scale: 0.8 }}
   505→                animate={{ opacity: 1, scale: 1 }}
   506→                exit={{ opacity: 0, scale: 0.8 }}
   507→                whileHover={{ scale: 1.05 }}
   508→                whileTap={{ scale: 0.95 }}
   509→                onClick={() => onAssignClick(pid)}
   510→                className="flex items-center gap-2 px-3 py-1.5 rounded-lg border border-amber-500/30 bg-card hover:bg-amber-500/10 hover:border-amber-500/50 transition-all duration-200 shrink-0 cursor-pointer group"
   511→              >
   512→                <div
   513→                  className="h-6 w-6 rounded-full flex items-center justify-center text-[10px] font-bold text-white"
   514→                  style={{ backgroundColor: p.color }}
   515→                >
   516→                  {getInitials(p.name)}
   517→                </div>
   518→                <span className="text-xs font-medium whitespace-nowrap">{p.name}</span>
   519→                <ArrowRight className="h-3 w-3 text-amber-500 opacity-0 group-hover:opacity-100 transition-opacity" />
   520→              </motion.button>
   521→            );
   522→          })}
   523→        </AnimatePresence>
   524→      </div>
   525→    </motion.div>
   526→  );
   527→}
