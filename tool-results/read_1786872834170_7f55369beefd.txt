    89→function useCountUp(target: number, duration = 1200, delay = 0) {
    90→  const [count, setCount] = useState(0)
    91→  useEffect(() => {
    92→    const startTime = performance.now() + delay
    93→    function step(now: number) {
    94→      if (now < startTime) {
    95→        requestAnimationFrame(step)
    96→        return
    97→      }
    98→      const elapsed = now - startTime
    99→      const progress = Math.min(elapsed / duration, 1)
   100→      const eased = 1 - Math.pow(1 - progress, 3)
   101→      setCount(Math.round(eased * target))
   102→      if (progress < 1) requestAnimationFrame(step)
   103→    }
   104→    requestAnimationFrame(step)
   105→  }, [target, duration, delay])
   106→  return count
   107→}
