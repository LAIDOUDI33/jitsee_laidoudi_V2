   106→  const { user } = useAppStore()

   108→  const [input, setInput] = useState('')

   122→    channels
   123→      .filter(c => c.name.toLowerCase().includes(searchQuery.toLowerCase()))

   126→        if (!groups[cat]) groups[cat] = []
   127→        groups[cat].push(c)
   128→      })

   130→  }, [searchQuery])
   131→
   132→  useEffect(() => {

   134→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   135→    }
   136→  }, [messages.length])
   137→
   138→  const handleSend = () => {
   139→    if (!input.trim()) return
   140→    const msg: ChatMessage = {
   141→      id: `msg-${Date.now()}`,
   142→      channelId: activeChannel,
   143→      senderId: user?.id || 'me',
   144→      senderName: user?.name || 'You',
   145→      content: input.trim(),
   146→      timestamp: new Date().toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' }),
   147→      status: 'sent',
   148→      reactions: {},
   149→    }
   150→    setLocalMessages(prev => ({
   151→      ...prev,
   152→      [activeChannel]: [...(prev[activeChannel] || []), msg],
   153→    }))
   154→    setInput('')
   155→  }
   156→
   157→  const handleReact = (msgId: string, emoji: string) => {
   158→    setLocalMessages(prev => {
   159→      const channelMsgs = prev[activeChannel] || []

   161→        ...prev,
   162→        [activeChannel]: channelMsgs.map(m => {
   163→          if (m.id !== msgId) return m

   165→          reactions[emoji] = (reactions[emoji] || 0) + 1
   166→          return { ...m, reactions }
   167→        }),
   168→      }
   169→    })
   170→    setShowReactions(null)
   171→  }
   172→
   173→  const handleCopyMessage = (content: string) => {
   174→    navigator.clipboard.writeText(content)
   175→    toast.success('Message copied!')
   176→  }
   177→

   179→    setCollapsedCategories(prev => {
   180→      const next = new Set(prev)
   181→      if (next.has(cat)) next.delete(cat)
   182→      else next.add(cat)
   183→      return next
   184→    })
   185→  }

   192→  }

   204→              <h3 className='font-semibold text-sm'>Channels</h3>

   233→                    )}
   234→                  </button>
   235→                  <AnimatePresence>
   236→                    {!isCollapsed && (
   237→                      <motion.div
   238→                        initial={{ height: 0, opacity: 0 }}
   239→                        animate={{ height: 'auto', opacity: 1 }}
   240→                        exit={{ height: 0, opacity: 0 }}
   241→                        transition={{ duration: 0.2 }}
   242→                        className='overflow-hidden space-y-0.5'
   243→                      >
   244→                        {chs.map(c => (

   249→                              activeChannel === c.id ? 'bg-primary/10 text-primary font-medium' : 'text-muted-foreground hover:bg-muted hover:text-foreground'

   257→                            )}
   258→                          </button>

   276→                <div className='relative'>

   279→                <span className='text-xs text-muted-foreground truncate group-hover:text-foreground transition-colors'>{u.name}</span>
   280→              </div>
   281→            ))}
   282→          </div>
   283→        </div>
   284→      </div>
   285→
   286→      {/* Chat area */}
   287→      <div className='flex-1 flex flex-col min-w-0'>
   288→        {/* Chat header */}
   289→        <div className='h-14 border-b flex items-center justify-between px-4 shrink-0 bg-card/50 backdrop-blur-sm'>
   290→          <div className='flex items-center gap-2.5'>
   291→            {channel?.type === 'announcement' ? <Megaphone className='h-5 w-5 text-amber-500' /> : <Hash className='h-5 w-5 text-muted-foreground' />}
   292→            <div>
   293→              <h3 className='font-semibold text-sm'>{channel?.name || 'general'}</h3>
   294→              <p className='text-[11px] text-muted-foreground'>{channel?.type || 'text'} channel</p>
   295→            </div>
   296→            {channel?.type === 'announcement' && (

   301→          </div>

   303→            <Button variant='ghost' size='icon' className='h-8 w-8 hover:scale-105 transition-transform'><Phone className='h-4 w-4' /></Button>

   305→              variant={showOnlinePanel ? 'secondary' : 'ghost'}

   308→              onClick={() => setShowOnlinePanel(!showOnlinePanel)}

   310→              {showOnlinePanel ? <PanelRightClose className='h-4 w-4' /> : <PanelRightOpen className='h-4 w-4' />}
   311→            </Button>
   312→            <Button variant='ghost' size='icon' className='h-8 w-8 hover:scale-105 transition-transform'><Pin className='h-4 w-4' /></Button>
   313→            <Button variant='ghost' size='icon' className='h-8 w-8 hover:scale-105 transition-transform'><MoreVertical className='h-4 w-4' /></Button>
   314→          </div>
   315→        </div>
   316→
   317→        {/* Messages */}
   318→        <div ref={scrollRef} className='flex-1 overflow-y-auto p-4 space-y-1'>
   319→          {messages.length === 0 && (
   320→            <div className='flex flex-col items-center justify-center py-16'>
   321→              <div className='relative'>
   322→                <Hash className='h-16 w-16 text-muted-foreground/20' />
   323→                <div className='absolute inset-0 flex items-center justify-center'>
   324→                  <Hash className='h-8 w-8 text-muted-foreground/40' />
   325→                </div>
   326→              </div>
   327→              <p className='font-medium mt-4'>No messages yet</p>
   328→              <p className='text-sm text-muted-foreground mt-1'>Start the conversation in #{channel?.name}</p>
   329→              <Button variant='outline' className='mt-4 gap-2 hover:scale-[1.02] active:scale-[0.98] transition-transform' onClick={() => document.querySelector<HTMLInputElement>('input[placeholder]')?.focus()}>
   330→                <Send className='h-4 w-4' /> Send First Message
   331→              </Button>
   332→            </div>

   354→                      <span className='text-sm font-semibold hover:underline cursor-pointer'>{msg.senderName}</span>

   409→                      <Tooltip>
   410→                        <TooltipTrigger asChild>

   412→                            <MoreVertical className='h-3.5 w-3.5 text-muted-foreground' />
