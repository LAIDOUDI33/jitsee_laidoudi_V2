   159→function formatTime(seconds: number): string {
   160→  const m = Math.floor(seconds / 60);
   161→  const s = seconds % 60;
   162→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;

   213→function AvatarStack({ participantIds, max = 4 }: { participantIds: string[]; max?: number }) {
   214→  const { getParticipant } = useBreakout();
   215→  const visible = participantIds.slice(0, max);
   216→  const remaining = participantIds.length - max;
   217→
   218→  return (
   219→    <div className="flex items-center">
   220→      {visible.map((pid, i) => {
   221→        const p = getParticipant(pid);
   222→        if (!p) return null;
   223→        return (
   224→          <TooltipProvider key={pid} delayDuration={0}>
   225→            <Tooltip>
   226→              <TooltipTrigger asChild>
   227→                <div
   228→                  className="h-7 w-7 rounded-full flex items-center justify-center text-[11px] font-bold text-white ring-2 ring-card -ml-[6px] first:ml-0 transition-transform hover:scale-110 hover:z-10 cursor-pointer"
   229→                  style={{ backgroundColor: p.color, zIndex: visible.length - i }}
   230→                >
   231→                  {getInitials(p.name)}
   232→                </div>
   233→              </TooltipTrigger>
   234→              <TooltipContent side="bottom" className="text-xs">
   235→                {p.name}
   236→              </TooltipContent>
   237→            </Tooltip>
   238→          </TooltipProvider>
   239→        );
   240→      })}
   241→      {remaining > 0 && (
   242→        <div className="h-7 w-7 rounded-full flex items-center justify-center text-[10px] font-semibold bg-muted text-muted-foreground ring-2 ring-card -ml-[6px]">
   243→          +{remaining}
   244→        </div>
   245→      )}
   246→    </div>
   247→  );
   248→}

   451→/* -------------------------------------------------------------------------- */
   452→/*                       BROADCAST MESSAGE DIALOG                             */
   453→/* -------------------------------------------------------------------------- */
   454→
   455→function BroadcastDialog({ open, onClose, targetRoom, rooms, onSend }: {
   456→  open: boolean;
   457→  onClose: () => void;
   458→  targetRoom: BreakoutRoom | null;
   459→  rooms: BreakoutRoom[];
   460→  onSend: (message: string, roomId: string | null) => void;
   461→}) {
   462→  const [message, setMessage] = useState('');
   463→  const [sendToRoom, setSendToRoom] = useState<string | null>(targetRoom?.id ?? null);
   464→
   465→  const handleSend = () => {
   466→    if (!message.trim()) return;
   467→    onSend(message.trim(), sendToRoom);
   468→    onClose();
   469→  };
   470→
   471→  return (
   472→    <Dialog open={open} onOpenChange={onClose}>
   473→      <DialogContent className="sm:max-w-md">
   474→        <DialogHeader>
   475→          <DialogTitle className="flex items-center gap-2.5">
   476→            <Volume2 className="h-5 w-5 text-amber-500" />
   477→            Broadcast Message
   478→          </DialogTitle>
   479→        </DialogHeader>
   480→
   481→        <div className="space-y-4 pt-1">
   482→          <div className="space-y-2">
   483→            <Label className="text-sm font-medium">Send To</Label>
   484→            <div className="flex flex-wrap gap-2">
   485→              <Button
   486→                variant={sendToRoom === null ? 'default' : 'outline'}
   487→                size="sm"
   488→                onClick={() => setSendToRoom(null)}
   489→                className={sendToRoom === null ? 'bg-gradient-to-r from-amber-500 to-orange-500 text-white border-0' : ''}
   490→              >
   491→                All Rooms
   492→              </Button>
   493→              {rooms.map((r) => (
   494→                <Button
   495→                  key={r.id}
   496→                  variant={sendToRoom === r.id ? 'default' : 'outline'}
   497→                  size="sm"
   498→                  onClick={() => setSendToRoom(r.id)}
   499→                  className={sendToRoom === r.id ? 'bg-gradient-to-r from-amber-500 to-orange-500 text-white border-0' : ''}
   500→                >
   501→                  {r.name}
   502→                </Button>
   503→              ))}
   504→            </div>
   505→          </div>
   506→
   507→          <div className="space-y-2">
   508→            <Label className="text-sm font-medium">Message</Label>
   509→            <textarea
   510→              className="flex min-h-[80px] w-full rounded-md border border-input bg-background px-3 py-2 text-sm ring-offset-background placeholder:text-muted-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-amber-500/30 focus-visible:ring-offset-2 resize-none"
   511→              placeholder="Type your broadcast message..."
   512→              value={message}
   513→              onChange={(e) => setMessage(e.target.value)}
   514→              onKeyDown={(e) => {
   515→                if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) handleSend();
   516→              }}
   517→            />
   518→          </div>
   519→
   520→          {message.trim() && (
   521→            <div className="space-y-1.5">
   522→              <p className="text-xs font-medium text-muted-foreground">Preview</p>

   650→  const selectedRoom = rooms.find((r) => r.id === selectedRoomId) || null;
   651→

   653→  useEffect(() => {
   654→    if (globalTimerMode === 'running') {

   656→        setRooms((prev) =>
   657→          prev.map((r) => {

   659→              const newElapsed = r.elapsedSeconds + 1;
   660→              let newStatus: RoomStatus = r.status === 'not-started' ? 'in-progress' : r.status;
   661→              if (settings.autoClose && settings.timeLimitMinutes > 0 && newElapsed >= settings.timeLimitMinutes * 60) {
   662→                newStatus = 'completed';

   665→                ...r,
   666→                elapsedSeconds: newElapsed,
   667→                status: newStatus,
   668→                timerMode: newStatus === 'completed' ? 'idle' : 'running',

   670→            }
   671→            return r;
   672→          })
   673→        );
   674→      }, 1000);
   675→    } else {

   714→          return r;
   715→        })
   716→      );
   717→      const p = getParticipant(pid);
   718→      toast.success(`${p?.name} moved to ${toRoom.name}`);
   719→    },

   736→        assigned++;
   737→      }

   832→          animate={{ opacity: 1, y: 0 }}

   995→function RoomCard({ room, index, onRoomClick, onBroadcast, onTimerToggle, onTimerReset }: {
   996→  room: BreakoutRoom;
   997→  index: number;
   998→  onRoomClick: (id: string) => void;
   999→  onBroadcast: (roomId: string) => void;
  1000→  onTimerToggle: (roomId: string) => void;

  1002→}) {
  1003→  const { settings } = useBreakout();

  1005→  const isOverTime = settings.timeLimitMinutes > 0 && room.elapsedSeconds >= settings.timeLimitMinutes * 60;
  1006→
  1007→  return (
