  1307→          >
  1308→            {isPinned ? <PinOff size={14} /> : <Pin size={14} />}
  1309→          </motion.button>
  1310→        )}
  1311→      </AnimatePresence>
  1312→
  1313→      {/* Name Label + Role Badge (bottom) */}
  1314→      <div className="absolute bottom-0 left-0 right-0 z-20 bg-gradient-to-t from-black/70 via-black/30 to-transparent pt-8 pb-2.5 px-3">
  1315→        <div className="flex items-center justify-between">
  1316→          <div className="flex items-center gap-2 min-w-0">
  1317→            <span className={`font-medium truncate ${compact ? 'text-[11px]' : 'text-sm'}`}>{participant.name}</span>
  1318→            <span className={`text-[10px] px-1.5 py-0.5 rounded-md font-medium border shrink-0 ${getRoleBadgeClass(participant.role)}`}>
  1319→              {participant.role}
  1320→            </span>
  1321→          </div>
  1322→          <div className="flex items-center gap-1.5 shrink-0">
  1323→            {!participant.micOn && <MicOff size={compact ? 12 : 14} className="text-red-400" />}
  1324→            {!participant.videoOn && <VideoOff size={compact ? 12 : 14} className="text-white/30" />}
  1325→          </div>
  1326→        </div>
  1327→      </div>
  1328→    </motion.div>
  1329→  );
  1330→}
