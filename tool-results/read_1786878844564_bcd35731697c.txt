    39→function RippleButton({ children, className, onClick }: { children: React.ReactNode; className?: string; onClick?: () => void }) {
    40→  const [ripples, setRipples] = useState<{ id: number; x: number; y: number }[]>([]);
    41→  const handleClick = useCallback((e: React.MouseEvent<HTMLButtonElement>) => {
    42→    const rect = e.currentTarget.getBoundingClientRect();
    43→    const x = e.clientX - rect.left;
    44→    const y = e.clientY - rect.top;
    45→    const id = Date.now();
    46→    setRipples(prev => [...prev, { id, x, y }]);
    47→    setTimeout(() => setRipples(prev => prev.filter(r => r.id !== id)), 600);
    48→    onClick?.();
    49→  }, [onClick]);
    50→  return (
    51→    <button onClick={handleClick} className={`relative overflow-hidden ${className || ''}`}>
    52→      {children}
    53→      {ripples.map(r => (
    54→        <motion.span
    55→          key={r.id}
    56→          initial={{ scale: 0, opacity: 0.35 }}
    57→          animate={{ scale: 4, opacity: 0 }}
    58→          transition={{ duration: 0.6, ease: 'easeOut' }}
    59→          className='absolute rounded-full bg-white/30 pointer-events-none'
    60→          style={{ left: r.x - 10, top: r.y - 10, width: 20, height: 20 }}
    61→        />
    62→      ))}
    63→    </button>
    64→  );
    65→}
