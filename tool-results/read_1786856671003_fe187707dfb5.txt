    94→function getCountdown(date: string, time: string): string | null {
    95→  if (!date) return null
    96→  const target = new Date(`${date}T${time === '9:00 AM' ? '09:00' : time}`)
    97→  const now = new Date()
    98→  const diff = target.getTime() - now.getTime()
    99→  if (diff <= 0) return null
   100→  const hours = Math.floor(diff / (1000 * 60 * 60))
   101→  const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   102→  if (hours > 24) return `Starts in ${Math.floor(hours / 24)}d ${hours % 24}h`
   103→  if (hours > 0) return `Starts in ${hours}h ${mins}m`
   104→  return `Starts in ${mins}m`
   105→}
