   236→// ─── Network Quality Indicator ────────────────────────────────
   237→function NetworkQualityIndicator() {
   238→  const [quality, setQuality] = useState<NetworkQuality>('excellent');
   239→  const [latency, setLatency] = useState(22);
   240→
   241→  useEffect(() => {
   242→    const qualities: NetworkQuality[] = ['excellent', 'good', 'fair', 'poor'];
   243→    const changeQuality = () => {
   244→      const q = qualities[Math.floor(Math.random() * qualities.length)];
   245→      setQuality(q);
   246→      const [minLat, maxLat] = networkQualityConfig[q].latency;
   247→      setLatency(Math.floor(Math.random() * (maxLat - minLat + 1)) + minLat);
   248→    };
   249→    const interval = setInterval(changeQuality, 10000 + Math.random() * 5000);
   250→    return () => clearInterval(interval);
   251→  }, []);
   252→
   253→  const config = networkQualityConfig[quality];
   254→  const barCount = quality === 'excellent' ? 4 : quality === 'good' ? 3 : quality === 'fair' ? 2 : 1;
   255→
   256→  return (
   257→    <Tooltip>
   258→      <TooltipTrigger asChild>
   259→        <div className="absolute top-4 right-4 z-30 flex items-center gap-2 px-2.5 py-1.5 rounded-xl bg-black/40 backdrop-blur-xl border border-white/10 cursor-default">
   260→          <div className="flex items-end gap-[2px] h-3.5">
   261→            {[1, 2, 3, 4].map((bar) => (
   262→              <motion.div
   263→                key={bar}
   264→                className={`w-[3px] rounded-full ${bar <= barCount ? config.barColor : 'bg-white/20'}`}
   265→                animate={{ height: [4, bar <= barCount ? bar * 3.5 : 4, bar <= barCount ? bar * 3.5 : 4] }}
   266→                transition={{ duration: 0.5, delay: bar * 0.05, ease: 'easeOut' }}
   267→              />
   268→            ))}
   269→          </div>
   270→          {quality === 'poor' ? <Signal size={12} className={config.color} /> : <Wifi size={12} className={config.color} />}
   271→        </div>
   272→      </TooltipTrigger>
   273→      <TooltipContent side="bottom" className="bg-slate-800/95 backdrop-blur-xl text-white border-white/10 text-xs rounded-lg">
   274→        <div className="flex flex-col gap-0.5">
   275→          <span>Network: <span className={config.color}>{config.label}</span></span>
   276→          <span className="text-white/50">Latency: {latency}ms</span>
   277→        </div>
   278→      </TooltipContent>
   279→    </Tooltip>
   280→  );
   281→}
