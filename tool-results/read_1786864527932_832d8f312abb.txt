   265→// ─── Network Quality Indicator ────────────────────────────────
   266→function NetworkQualityIndicator() {
   267→  const [quality, setQuality] = useState<NetworkQuality>('excellent');
   268→  const [latency, setLatency] = useState(22);
   269→
   270→  useEffect(() => {
   271→    const qualities: NetworkQuality[] = ['excellent', 'good', 'fair', 'poor'];
   272→    const changeQuality = () => {
   273→      const q = qualities[Math.floor(Math.random() * qualities.length)];
   274→      setQuality(q);
   275→      const [minLat, maxLat] = networkQualityConfig[q].latency;
   276→      setLatency(Math.floor(Math.random() * (maxLat - minLat + 1)) + minLat);
   277→    };
   278→    const interval = setInterval(changeQuality, 10000 + Math.random() * 5000);
   279→    return () => clearInterval(interval);
   280→  }, []);
   281→
   282→  const config = networkQualityConfig[quality];
   283→  const barCount = quality === 'excellent' ? 4 : quality === 'good' ? 3 : quality === 'fair' ? 2 : 1;
   284→
   285→  return (
   286→    <Tooltip>
   287→      <TooltipTrigger asChild>
   288→        <div className="absolute top-4 right-4 z-30 flex items-center gap-2 px-2.5 py-1.5 rounded-xl bg-black/40 backdrop-blur-xl border border-white/10 cursor-default">
   289→          <div className="flex items-end gap-[2px] h-3.5">
   290→            {[1, 2, 3, 4].map((bar) => (
   291→              <motion.div
   292→                key={bar}
   293→                className={`w-[3px] rounded-full ${bar <= barCount ? config.barColor : 'bg-white/20'}`}
   294→                animate={{ height: [4, bar <= barCount ? bar * 3.5 : 4, bar <= barCount ? bar * 3.5 : 4] }}
   295→                transition={{ duration: 0.5, delay: bar * 0.05, ease: 'easeOut' }}
   296→              />
   297→            ))}
   298→          </div>
   299→          {quality === 'poor' ? <Signal size={12} className={config.color} /> : <Wifi size={12} className={config.color} />}
   300→        </div>
   301→      </TooltipTrigger>
   302→      <TooltipContent side="bottom" className="bg-slate-800/95 backdrop-blur-xl text-white border-white/10 text-xs rounded-lg">
   303→        <div className="flex flex-col gap-0.5">
   304→          <span>Network: <span className={config.color}>{config.label}</span></span>
   305→          <span className="text-white/50">Latency: {latency}ms</span>
   306→        </div>
   307→      </TooltipContent>
   308→    </Tooltip>
   309→  );
   310→}
