   380→function generateRoomId(): string {
   381→  const group = () =>
   382→    Array.from({ length: 3 }, () =>
   383→      String.fromCharCode(97 + Math.floor(Math.random() * 26)),
   384→    ).join('');
   385→  return `${group()}-${group()}-${group()}`;
   386→}

   389→/*                          ANIMATED COUNTER COMPONENT                       */
   390→/* -------------------------------------------------------------------------- */
   391→
   392→function AnimatedCounter({ target, suffix, label }: StatItem) {
   393→  const ref = useRef<HTMLSpanElement>(null);
   394→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   395→  const [count, setCount] = useState(0);
   396→
   397→  useEffect(() => {
   398→    if (!isInView) return;
   399→
   400→    const duration = 2000;
   401→    const steps = 60;
   402→    const increment = target / steps;
   403→    let current = 0;
   404→    const timer = setInterval(() => {
   405→      current += increment;
   406→      if (current >= target) {
   407→        setCount(target);
   408→        clearInterval(timer);
   409→      } else {
   410→        setCount(current);
   411→      }
   412→    }, duration / steps);
   413→
   414→    return () => clearInterval(timer);
   415→  }, [isInView, target]);
   416→
   417→  const displayValue =
   418→    target % 1 !== 0 ? count.toFixed(1) : Math.round(count).toLocaleString();
   419→
   420→  return (
   421→    <div className="text-center">
   422→      <div className="text-4xl md:text-5xl font-bold tracking-tight">
   423→        <span ref={ref}>{displayValue}</span>
   424→        <span className="gradient-text">{suffix}</span>
   425→      </div>
   426→      <p className="mt-2 text-muted-foreground text-sm font-medium">{label}</p>
   427→    </div>
   428→  );
   429→}
