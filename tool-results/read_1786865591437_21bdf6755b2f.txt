   144→function StatusIcon({ status }: { status?: string }) {
   145→  if (status === 'read') return <CheckCheck className='h-3.5 w-3.5 text-primary' />
   146→  if (status === 'delivered') return <CheckCheck className='h-3.5 w-3.5 text-muted-foreground' />
   147→  if (status === 'sent') return <Check className='h-3.5 w-3.5 text-muted-foreground' />
   148→  return null

   558→                        initial={{ opacity: 0, scale: 0.9 }}

   560→                        exit={{ opacity: 0, scale: 0.9 }}

   574→              </div>

   577→          {/* Typing indicator */}

   638→                <Users className='h-4 w-4' />
   639→                Members
   640→              </h4>
   641→            </div>
   642→            <ScrollArea className='flex-1'>
   643→              <div className='p-2 space-y-1'>
   644→                <p className='text-[10px] font-semibold text-muted-foreground/60 uppercase tracking-wider px-1 mb-1'>
   645→                  Online — {effectiveOnlineUsers.filter(u => u.status === 'active').length}
   646→                </p>
   647→                {effectiveOnlineUsers.filter(u => u.status === 'active').map(u => (
   648→                  <div key={u.name} className='flex items-center gap-2.5 px-2 py-1.5 rounded-lg hover:bg-muted/50 transition-colors cursor-pointer'>
   649→                    <div className='relative'>
   650→                      <Avatar className='h-7 w-7'>
