    97→function generateSecret(): string {
    98→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';
    99→  let result = 'whsec_';
   100→  for (let i = 0; i < 32; i++) result += chars.charAt(Math.floor(Math.random() * chars.length));
   101→  return result;
   102→}
   103→
   104→function maskSecret(secret: string): string {
   105→  if (secret.length <= 12) return '••••••••••••';
   106→  return secret.slice(0, 7) + '••••••••••••' + secret.slice(-4);
   107→}
   108→
   109→function truncateUrl(url: string, maxLen = 40): string {
   110→  if (url.length <= maxLen) return url;
   111→  return url.slice(0, maxLen - 3) + '...';
   112→}

   170→function StatusBadge({ code }: { code: number }) {
   171→  const color =
   172→    code >= 200 && code < 300
   173→      ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-500/15 dark:text-emerald-400'
   174→      : code >= 400 && code < 500
   175→        ? 'bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400'
   176→        : 'bg-red-100 text-red-700 dark:bg-red-500/15 dark:text-red-400';
   177→  return (
   178→    <Badge variant="secondary" className={`text-[11px] font-mono font-medium px-2 py-0.5 ${color}`}>
   179→      {code}
   180→    </Badge>
   181→  );
   182→}
