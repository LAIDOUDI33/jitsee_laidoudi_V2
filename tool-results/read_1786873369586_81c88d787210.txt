   582→export default function NotificationsPage() {
   583→  const [notifications, setNotifications] = useState<Notification[]>(initialNotifications)
   584→  const [activeFilter, setActiveFilter] = useState<NotificationCategory>('all')
   585→  const [activeTab, setActiveTab] = useState<TabCategory>('recent')
   586→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   587→  const [pendingDeleteId, setPendingDeleteId] = useState<string | null>(null)
   588→
   589→  // Count unread
   590→  const unreadCount = useMemo(
   591→    () => notifications.filter(n => n.unread).length,
   592→    [notifications]
   593→  )
   594→
   595→  // Filter by category (filter pills)
   596→  const categoryFiltered = useMemo(() => {
   597→    switch (activeFilter) {
   598→      case 'unread':
   599→        return notifications.filter(n => n.unread)
   600→      case 'mentions':
   601→        return notifications.filter(n => n.type === 'mention')
   602→      case 'meetings':
   603→        return notifications.filter(n => ['meeting-invite', 'meeting-soon', 'recording-ready'].includes(n.type))
   604→      case 'system':
   605→        return notifications.filter(n => ['security-alert', 'maintenance', 'system-update'].includes(n.type))
   606→      default:
   607→        return notifications
   608→    }
   609→  }, [notifications, activeFilter])
   610→
   611→  // Filter by tab
   612→  const tabFiltered = useMemo(() => {
   613→    switch (activeTab) {
   614→      case 'mentions':
   615→        return categoryFiltered.filter(n => n.type === 'mention')
   616→      case 'meeting-alerts':
   617→        return categoryFiltered.filter(n => ['meeting-invite', 'meeting-soon', 'recording-ready', 'ai-summary'].includes(n.type))
   618→      case 'system-updates':
   619→        return categoryFiltered.filter(n => ['security-alert', 'maintenance', 'system-update', 'member-joined'].includes(n.type))
   620→      default:
   621→        return categoryFiltered
   622→    }
   623→  }, [categoryFiltered, activeTab])
   624→
   625→  // Sort: pinned first, then by time group, then by original order
   626→  const sorted = useMemo(() => {
   627→    const groupOrder: Record<TimeGroup, number> = { 'Today': 0, 'Yesterday': 1, 'Earlier': 2 }
   628→    return [...tabFiltered].sort((a, b) => {
   629→      if (a.pinned !== b.pinned) return a.pinned ? -1 : 1
   630→      if (groupOrder[a.timeGroup] !== groupOrder[b.timeGroup]) {
   631→        return groupOrder[a.timeGroup] - groupOrder[b.timeGroup]
   632→      }
   633→      return 0
   634→    })
   635→  }, [tabFiltered])
   636→
   637→  // Group by time for 'recent' tab
   638→  const grouped = useMemo(() => {
   639→    if (activeTab !== 'recent') return { 'All': sorted }
   640→    const groups: Record<TimeGroup, Notification[]> = { 'Today': [], 'Yesterday': [], 'Earlier': [] }
   641→    sorted.forEach(n => groups[n.timeGroup].push(n))
   642→    return groups
   643→  }, [sorted, activeTab])
   644→
   645→  // Handlers
   646→  const handleMarkRead = (id: string) => {
   647→    setNotifications(prev => prev.map(n => n.id === id ? { ...n, unread: false } : n))
   648→  }

   651→    const unreadCount = notifications.filter(n => n.unread).length
   652→    if (unreadCount === 0) {
   653→      toast.info('All notifications are already read')
   654→      return
   655→    }

   658→  }
   659→
   660→  const handleDelete = (id: string) => {

   662→    setDeleteDialogOpen(true)

   669→    setPendingDeleteId(null)
   670→    toast.success('Notification deleted')
   671→  }
   672→
   673→  const handleTogglePin = (id: string) => {
   674→    setNotifications(prev => prev.map(n => n.id === id ? { ...n, pinned: !n.pinned } : n))
   675→    const n = notifications.find(n => n.id === id)
   676→    if (n) toast.success(n.pinned ? 'Notification unpinned' : 'Notification pinned')
   677→  }
   678→
   679→  const handleAction = (id: string, actionLabel: string) => {
   680→    toast.success(actionLabel)
   681→  }
   682→
   683→  const activeFilterLabel = filterPills.find(p => p.value === activeFilter)?.label ?? ''
   684→
   685→  return (
   686→    <>

   688→        className='space-y-6'

   690→        initial='hidden'

   695→          <div className='flex items-center gap-3'>

   697→              <Bell className='h-6 w-6' />
   698→            </div>
   699→            <div>
   700→              <h1 className='text-2xl font-bold tracking-tight'>Notifications</h1>

   702→            </div>
   703→          </div>
   704→          <Button
   705→            variant='outline'
   706→            className='shrink-0 gap-2'
   707→            onClick={handleMarkAllRead}
   708→          >
   709→            <CheckCheck className='h-4 w-4' />
   710→            Mark All Read
   711→          </Button>
   712→        </motion.div>
   713→
   714→        {/* ── Filter Pills ── */}
   715→        <motion.div variants={item} className='flex flex-wrap items-center gap-2'>

   730→                  <span className={`ml-1.5 inline-flex items-center justify-center w-4 h-4 rounded-full text-[10px] font-bold ${
