    96→function getCountdown(date: string, time: string): string | null {
    97→  if (!date) return null
    98→  const target = new Date(`${date}T${time === '9:00 AM' ? '09:00' : time}`)
    99→  const now = new Date()
   100→  const diff = target.getTime() - now.getTime()
   101→  if (diff <= 0) return null
   102→  const hours = Math.floor(diff / (1000 * 60 * 60))
   103→  const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))
   104→  if (hours > 24) return `Starts in ${Math.floor(hours / 24)}d ${hours % 24}h`
   105→  if (hours > 0) return `Starts in ${hours}h ${mins}m`
   106→  return `Starts in ${mins}m`
   107→}
