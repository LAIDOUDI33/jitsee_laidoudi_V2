    95→export default function WhiteboardPage() {
    96→  const { user } = useAppStore()

    98→  // Canvas refs
    99→  const canvasRef = useRef<HTMLCanvasElement>(null)
   100→  const minimapRef = useRef<HTMLCanvasElement>(null)
   101→
   102→  // Tool state
   103→  const [activeTool, setActiveTool] = useState<ToolType>('pen')
   104→  const [activeColor, setActiveColor] = useState(COLORS[0])
   105→  const [activeWidth, setActiveWidth] = useState(STROKE_WIDTHS[1])
   106→  const [showGrid, setShowGrid] = useState(true)
   107→  const [zoom, setZoom] = useState(1)
   108→  const [isDrawing, setIsDrawing] = useState(false)
   109→
   110→  // History state
   111→  const [history, setHistory] = useState<DrawAction[][]>([[]])
   112→  const [historyIndex, setHistoryIndex] = useState(0)
   113→  const [currentAction, setCurrentAction] = useState<DrawAction | null>(null)
   114→
   115→  // Canvas dimensions
   116→  const [canvasSize, setCanvasSize] = useState({ width: 1920, height: 1080 })
   117→  const containerRef = useRef<HTMLDivElement>(null)
   118→
   119→  // Text input state
   120→  const [textInput, setTextInput] = useState<{ x: number; y: number } | null>(null)
   121→  const [textValue, setTextValue] = useState('')
   122→
   123→  // Collaborator cursor animation
   124→  const [collaborators, setCollaborators] = useState<Collaborator[]>(MOCK_COLLABORATORS)
   125→
   126→  // ─── Resize handler ──────────────────────────────────────────────────────
   127→  useEffect(() => {
   128→    const handleResize = () => {
   129→      if (containerRef.current) {
   130→        setCanvasSize({
   131→          width: containerRef.current.clientWidth,
   132→          height: containerRef.current.clientHeight,
   133→        })
   134→      }
   135→    }
   136→    handleResize()
   137→    window.addEventListener('resize', handleResize)
   138→    return () => window.removeEventListener('resize', handleResize)
   139→  }, [])
   140→
   141→  // ─── Draw single action ──────────────────────────────────────────────────
   142→  const drawAction = useCallback(
   143→    (ctx: CanvasRenderingContext2D, action: DrawAction) => {
   144→      if (!action || action.points.length === 0) return
   145→
   146→      ctx.save()
   147→      ctx.strokeStyle = action.color
   148→      ctx.lineWidth = action.width * zoom
   149→      ctx.lineCap = 'round'
   150→      ctx.lineJoin = 'round'
   151→
   152→      if (action.type === 'eraser') {
   153→        ctx.globalCompositeOperation = 'destination-out'
   154→        ctx.strokeStyle = 'rgba(0,0,0,1)'
   155→      }
   156→
   157→      if (action.type === 'pen' || action.type === 'eraser' || action.type === 'select') {
   158→        ctx.beginPath()
   159→        ctx.moveTo(action.points[0].x * zoom, action.points[0].y * zoom)
   160→        for (let i = 1; i < action.points.length; i++) {
   161→          const prev = action.points[i - 1]
   162→          const curr = action.points[i]
   163→          const mx = ((prev.x + curr.x) / 2) * zoom
   164→          const my = ((prev.y + curr.y) / 2) * zoom
   165→          ctx.quadraticCurveTo(prev.x * zoom, prev.y * zoom, mx, my)
   166→        }
   167→        ctx.stroke()
   168→      } else if (action.type === 'rectangle') {
   169→        const start = action.points[0]
   170→        const end = action.points[action.points.length - 1]
   171→        ctx.strokeRect(
   172→          start.x * zoom,
   173→          start.y * zoom,
   174→          (end.x - start.x) * zoom,
   175→          (end.y - start.y) * zoom
   176→        )
   177→      } else if (action.type === 'circle') {
   178→        const start = action.points[0]
   179→        const end = action.points[action.points.length - 1]
   180→        const rx = Math.abs((end.x - start.x) / 2) * zoom
   181→        const ry = Math.abs((end.y - start.y) / 2) * zoom
   182→        const cx = ((start.x + end.x) / 2) * zoom
   183→        const cy = ((start.y + end.y) / 2) * zoom
   184→        ctx.beginPath()
   185→        ctx.ellipse(cx, cy, rx, ry, 0, 0, Math.PI * 2)
   186→        ctx.stroke()
   187→      } else if (action.type === 'line') {
   188→        const start = action.points[0]
   189→        const end = action.points[action.points.length - 1]
   190→        ctx.beginPath()
   191→        ctx.moveTo(start.x * zoom, start.y * zoom)
   192→        ctx.lineTo(end.x * zoom, end.y * zoom)
   193→        ctx.stroke()
   194→      } else if (action.type === 'arrow') {
   195→        const start = action.points[0]
   196→        const end = action.points[action.points.length - 1]
   197→        const sx = start.x * zoom
   198→        const sy = start.y * zoom
   199→        const ex = end.x * zoom
   200→        const ey = end.y * zoom
   201→        const angle = Math.atan2(ey - sy, ex - sx)
   202→        const headLen = 14 * zoom
   203→        ctx.beginPath()
   204→        ctx.moveTo(sx, sy)
   205→        ctx.lineTo(ex, ey)
   206→        ctx.stroke()
   207→        ctx.beginPath()
   208→        ctx.moveTo(ex, ey)
   209→        ctx.lineTo(ex - headLen * Math.cos(angle - Math.PI / 6), ey - headLen * Math.sin(angle - Math.PI / 6))
   210→        ctx.moveTo(ex, ey)
   211→        ctx.lineTo(ex - headLen * Math.cos(angle + Math.PI / 6), ey - headLen * Math.sin(angle + Math.PI / 6))
   212→        ctx.stroke()
   213→      } else if (action.type === 'text' && action.text) {
   214→        ctx.fillStyle = action.color
   215→        ctx.font = `${action.width * 4 * zoom}px Inter, system-ui, sans-serif`
   216→        ctx.fillText(action.text, action.points[0].x * zoom, action.points[0].y * zoom)
   217→      }
   218→      ctx.restore()
   219→    },
   220→    [zoom]
   221→  )
   222→
   223→  // ─── Redraw canvas ──────────────────────────────────────────────────────
   224→  const redrawCanvas = useCallback(
   225→    (ctx: CanvasRenderingContext2D, actions: DrawAction[][]) => {
   226→      const w = canvasSize.width
   227→      const h = canvasSize.height
   228→      ctx.clearRect(0, 0, w, h)
   229→
   230→      // Draw grid
   231→      if (showGrid) {
   232→        ctx.save()
   233→        ctx.fillStyle = '#94a3b8'
   234→        const spacing = 24 * zoom
   235→        for (let x = 0; x < w; x += spacing) {
   236→          for (let y = 0; y < h; y += spacing) {
   237→            ctx.beginPath()
   238→            ctx.arc(x, y, 1 * zoom, 0, Math.PI * 2)
   239→            ctx.fill()
   240→          }
   241→        }
   242→        ctx.restore()
   243→      }
   244→
   245→      // Draw all actions
   246→      for (const action of actions.flat()) {
   247→        drawAction(ctx, action)
   248→      }
   249→    },
   250→    [canvasSize, showGrid, zoom, drawAction]
   251→  )
   252→
   253→  // ─── Canvas effect ───────────────────────────────────────────────────────
   254→  useEffect(() => {
   255→    const canvas = canvasRef.current
   256→    if (!canvas) return
   257→    const ctx = canvas.getContext('2d')
   258→    if (!ctx) return
   259→
   260→    canvas.width = canvasSize.width
   261→    canvas.height = canvasSize.height
   262→
   263→    const snapshot = [...history.slice(0, historyIndex + 1)]
   264→    redrawCanvas(ctx, snapshot)
   265→
   266→    // Draw current action being drawn
   267→    if (currentAction) {
   268→      drawAction(ctx, currentAction)
   269→    }
   270→  }, [canvasSize, history, historyIndex, currentAction, redrawCanvas])
   271→
   272→  // ─── Minimap effect ──────────────────────────────────────────────────────
   273→  useEffect(() => {
   274→    const minimap = minimapRef.current
   275→    const canvas = canvasRef.current
   276→    if (!minimap || !canvas) return
   277→    const mctx = minimap.getContext('2d')
   278→    if (!mctx) return
   279→
   280→    const mw = 180
   281→    const mh = Math.round((mw / canvasSize.width) * canvasSize.height)
   282→    minimap.width = mw
   283→    minimap.height = mh
   284→    minimap.style.height = `${mh}px`
   285→
   286→    mctx.fillStyle = '#ffffff'
   287→    mctx.fillRect(0, 0, mw, mh)
   288→    mctx.drawImage(canvas, 0, 0, mw, mh)
   289→    mctx.strokeStyle = '#94a3b8'
   290→    mctx.lineWidth = 1
   291→    mctx.strokeRect(0, 0, mw, mh)
   292→  }, [canvasSize, history, historyIndex, currentAction])
   293→
   294→  // ─── Get position from mouse/touch event ─────────────────────────────────
   295→  function getPosition(e: React.MouseEvent<HTMLCanvasElement> | React.TouchEvent<HTMLCanvasElement>) {
   296→    const canvas = canvasRef.current
   297→    if (!canvas) return { x: 0, y: 0 }
   298→    const rect = canvas.getBoundingClientRect()
   299→    let clientX: number, clientY: number
   300→    if ('touches' in e) {
   301→      clientX = e.touches[0].clientX
   302→      clientY = e.touches[0].clientY
   303→    } else {
   304→      clientX = e.clientX
   305→      clientY = e.clientY
   306→    }
   307→    return {
   308→      x: (clientX - rect.left) / zoom,
   309→      y: (clientY - rect.top) / zoom,
   310→    }
   311→  }
   312→
   313→  // ─── Mouse / Touch handlers ─────────────────────────────────────────────
   314→  function handlePointerDown(e: React.MouseEvent<HTMLCanvasElement> | React.TouchEvent<HTMLCanvasElement>) {
   315→    const pos = getPosition(e)
   316→
   317→    if (activeTool === 'text') {
   318→      setTextInput({ x: pos.x, y: pos.y })
   319→      setTextValue('')
   320→      return
   321→    }
   322→
   323→    setIsDrawing(true)
   324→    const action: DrawAction = {
   325→      type: activeTool,
   326→      points: [pos],
   327→      color: activeColor,
   328→      width: activeWidth,
   329→    }
   330→    setCurrentAction(action)
   331→  }
   332→
   333→  function handlePointerMove(e: React.MouseEvent<HTMLCanvasElement> | React.TouchEvent<HTMLCanvasElement>) {
   334→    if (!isDrawing || !currentAction) return
   335→    const pos = getPosition(e)
   336→    setCurrentAction({ ...currentAction, points: [...currentAction.points, pos] })
   337→  }
   338→
   339→  function handlePointerUp() {
   340→    if (!isDrawing || !currentAction) {
   341→      setIsDrawing(false)
   342→      return
   343→    }
   344→    setIsDrawing(false)
   345→
   346→    if (currentAction.points.length > 0) {
   347→      const newHistory = [...history.slice(0, historyIndex + 1), [...history[historyIndex], currentAction]]
   348→      setHistory(newHistory)
   349→      setHistoryIndex(newHistory.length - 1)
   350→    }
   351→    setCurrentAction(null)
   352→  }
   353→
   354→  // ─── Text submit handler ──────────────────────────────────────────────────
   355→  function handleTextSubmit() {
   356→    if (!textInput || !textValue.trim()) {
   357→      setTextInput(null)
   358→      return
   359→    }
   360→    const action: DrawAction = {
   361→      type: 'text',
   362→      points: [{ x: textInput.x, y: textInput.y }],
   363→      color: activeColor,
   364→      width: activeWidth,
   365→      text: textValue,
   366→    }
   367→    const newHistory = [...history.slice(0, historyIndex + 1), [...history[historyIndex], action]]
   368→    setHistory(newHistory)
   369→    setHistoryIndex(newHistory.length - 1)
   370→    setTextInput(null)
   371→    setTextValue('')
   372→  }
   373→
   374→  // ─── Undo / Redo ─────────────────────────────────────────────────────────
   375→  function handleUndo() {
   376→    if (historyIndex > 0) {
   377→      setHistoryIndex(historyIndex - 1)
   378→    }
   379→  }
   380→
   381→  function handleRedo() {
   382→    if (historyIndex < history.length - 1) {
   383→      setHistoryIndex(historyIndex + 1)
   384→    }
   385→  }
   386→
   387→  // ─── Clear canvas ─────────────────────────────────────────────────────────
   388→  function handleClear() {
   389→    setHistory([[]])
   390→    setHistoryIndex(0)
   391→    setCurrentAction(null)
   392→  }
   393→
   394→  // ─── Zoom controls ───────────────────────────────────────────────────────
   395→  function handleZoomIn() {
   396→    setZoom((z) => Math.min(z + 0.1, 3))
   397→  }
   398→
   399→  function handleZoomOut() {
   400→    setZoom((z) => Math.max(z - 0.1, 0.3))
   401→  }
   402→
   403→  function handleZoomFit() {
   404→    setZoom(1)
   405→  }
   406→
   407→  // ─── Collaborator cursor animation ────────────────────────────────────────
   408→  useEffect(() => {
   409→    const interval = setInterval(() => {
   410→      setCollaborators((prev) =>
   411→        prev.map((c) => ({
   412→          ...c,
   413→          x: c.x + (Math.random() - 0.5) * 30,
   414→          y: c.y + (Math.random() - 0.5) * 30,
   415→        }))
   416→      )
   417→    }, 2000)
   418→    return () => clearInterval(interval)
   419→  }, [])
   420→
   421→  // ─── Export canvas ───────────────────────────────────────────────────────
   422→  function handleExport() {
   423→    const canvas = canvasRef.current
   424→    if (!canvas) return

   426→    link.download = 'alvision-whiteboard.png'
   427→    link.href = canvas.toDataURL()
   428→    link.click()
   429→  }

   439→          animate={{ y: 0, opacity: 1 }}
   440→          transition={{ type: 'spring', stiffness: 300, damping: 25 }}

   451→                    className={`h-8 w-8 ${
   452→                      activeTool === item.tool

   454→                        : 'hover:bg-muted/80'
   455→                    }`}
   456→                    onClick={() => setActiveTool(item.tool)}
   457→                  >
   458→                    {item.icon}
   459→                    <span className='sr-only'>{item.label}</span>
   460→                  </Button>
   461→                </TooltipTrigger>
   462→                <TooltipContent side='bottom' className='text-xs'>
   463→                  {item.label}
   464→                </TooltipContent>
   465→              </Tooltip>
   466→            ))}
   467→          </div>
   468→
   469→          <Separator orientation='vertical' className='mx-2 h-6' />
   470→
   471→          {/* Color picker */}
   472→          <div className='flex items-center gap-1'>
   473→            {COLORS.map((color) => (
   474→              <button
   475→                key={color}
   476→                className={`h-6 w-6 rounded-full border-2 transition-all duration-150 hover:scale-110 ${
   477→                  activeColor === color
   478→                    ? 'border-foreground ring-2 ring-primary/30 scale-110'
   479→                    : 'border-transparent hover:border-muted-foreground/50'
   480→                }`}
   481→                style={{ backgroundColor: color }}
   482→                onClick={() => setActiveColor(color)}
   483→                aria-label={`Color ${color}`}
   484→              />
   485→            ))}
   486→          </div>
   487→
   488→          <Separator orientation='vertical' className='mx-2 h-6' />
   489→
   490→          {/* Stroke width */}
   491→          <div className='flex items-center gap-1'>
   492→            {STROKE_WIDTHS.map((w) => (
   493→              <Tooltip key={w}>
   494→                <TooltipTrigger asChild>
   495→                  <Button
   496→                    variant={activeWidth === w ? 'default' : 'ghost'}
   497→                    size='icon'

   499→                      activeWidth === w
   500→                        ? 'shadow-md shadow-primary/20'
   501→                        : 'hover:bg-muted/80'
   502→                    }`}
   503→                    onClick={() => setActiveWidth(w)}
   504→                  >
   505→                    <span
   506→                      className='rounded-full bg-current'
   507→                      style={{
   508→                        width: `${Math.min(w * 2, 14)}px`,
   509→                        height: `${Math.min(w * 2, 14)}px`,
   510→                      }}
   511→                    />
   512→                    <span className='sr-only'>{w}px stroke</span>
   513→                  </Button>
   514→                </TooltipTrigger>
   515→                <TooltipContent side='bottom' className='text-xs'>
   516→                  {w}px
   517→                </TooltipContent>
   518→              </Tooltip>
   519→            ))}
   520→          </div>
   521→
   522→          <Separator orientation='vertical' className='mx-2 h-6' />
   523→
   524→          {/* Undo / Redo / Clear */}
   525→          <div className='flex items-center gap-0.5'>
   526→            <Tooltip>
   527→              <TooltipTrigger asChild>
   528→                <Button
   529→                  variant='ghost'
   530→                  size='icon'
   531→                  className='h-8 w-8 hover:bg-muted/80'
   532→                  onClick={handleUndo}
   533→                  disabled={historyIndex === 0}
   534→                >
   535→                  <Undo2 className='h-4 w-4' />
   536→                  <span className='sr-only'>Undo</span>
   537→                </Button>
   538→              </TooltipTrigger>
   539→              <TooltipContent side='bottom' className='text-xs'>Undo</TooltipContent>
   540→            </Tooltip>
   541→            <Tooltip>
   542→              <TooltipTrigger asChild>
   543→                <Button
   544→                  variant='ghost'
   545→                  size='icon'
   546→                  className='h-8 w-8 hover:bg-muted/80'
   547→                  onClick={handleRedo}
   548→                  disabled={historyIndex >= history.length - 1}
   549→                >

   551→                  <span className='sr-only'>Redo</span>

   556→            <Tooltip>
   557→              <TooltipTrigger asChild>
   558→                <Button

   571→
   572→          <Separator orientation='vertical' className='mx-2 h-6' />
   573→
   574→          {/* Zoom controls */}
   575→          <div className='flex items-center gap-0.5'>
