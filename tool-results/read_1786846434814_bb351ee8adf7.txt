   308→   308→function generateRoomId(): string {
   309→   309→  const group = () =>
   310→   310→    Array.from({ length: 3 }, () =>
   311→   311→      String.fromCharCode(97 + Math.floor(Math.random() * 26)),
   312→   312→    ).join('');
   313→   313→  return `${group()}-${group()}-${group()}`;
   314→   314→}

   317→   317→/*                          ANIMATED COUNTER COMPONENT                       */
   318→   318→/* -------------------------------------------------------------------------- */
   319→   319→
   320→   320→function AnimatedCounter({ target, suffix, label }: StatItem) {
   321→   321→  const ref = useRef<HTMLSpanElement>(null);
   322→   322→  const isInView = useInView(ref, { once: true, margin: '-50px' });
   323→   323→  const [count, setCount] = useState(0);
   324→   324→
   325→   325→  useEffect(() => {
   326→   326→    if (!isInView) return;
   327→   327→
   328→   328→    const duration = 2000;
   329→   329→    const steps = 60;
   330→   330→    const increment = target / steps;
   331→   331→    let current = 0;
   332→   332→    const timer = setInterval(() => {
   333→   333→      current += increment;
   334→   334→      if (current >= target) {
   335→   335→        setCount(target);
   336→   336→        clearInterval(timer);
   337→   337→      } else {
   338→   338→        setCount(current);
   339→   339→      }
   340→   340→    }, duration / steps);
   341→   341→
   342→   342→    return () => clearInterval(timer);
   343→   343→  }, [isInView, target]);
   344→   344→
   345→   345→  const displayValue =
   346→   346→    target % 1 !== 0 ? count.toFixed(1) : Math.round(count).toLocaleString();
   347→   347→
   348→   348→  return (
   349→   349→    <div className="text-center">
   350→   350→      <div className="text-4xl md:text-5xl font-bold tracking-tight">
   351→   351→        <span ref={ref}>{displayValue}</span>
   352→   352→        <span className="gradient-text">{suffix}</span>
   353→   353→      </div>
   354→   354→      <p className="mt-2 text-muted-foreground text-sm font-medium">{label}</p>
   355→   355→    </div>
   356→   356→  );
   357→   357→}
