    38→function RippleButton({ children, className, onClick }: { children: React.ReactNode; className?: string; onClick?: () => void }) {
    39→  const [ripples, setRipples] = useState<{ id: number; x: number; y: number }[]>([]);
    40→  const handleClick = useCallback((e: React.MouseEvent<HTMLButtonElement>) => {
    41→    const rect = e.currentTarget.getBoundingClientRect();
    42→    const x = e.clientX - rect.left;
    43→    const y = e.clientY - rect.top;
    44→    const id = Date.now();
    45→    setRipples(prev => [...prev, { id, x, y }]);
    46→    setTimeout(() => setRipples(prev => prev.filter(r => r.id !== id)), 600);
    47→    onClick?.();
    48→  }, [onClick]);
    49→  return (
    50→    <button onClick={handleClick} className={`relative overflow-hidden ${className || ''}`}>
    51→      {children}
    52→      {ripples.map(r => (
    53→        <motion.span
    54→          key={r.id}
    55→          initial={{ scale: 0, opacity: 0.35 }}
    56→          animate={{ scale: 4, opacity: 0 }}
    57→          transition={{ duration: 0.6, ease: 'easeOut' }}
    58→          className='absolute rounded-full bg-white/30 pointer-events-none'
    59→          style={{ left: r.x - 10, top: r.y - 10, width: 20, height: 20 }}
    60→        />
    61→      ))}
    62→    </button>
    63→  );
    64→}
